    12→//! Twoyi kernel-replacement daemon — the Rust port of VM's `libkr64.so`.
    13→//!
    14→//! See `VM_KR64_ANALYSIS.md` and `GSI_BOOT_PLAN.md` for the full
    15→//! background. The short version: twoyi needs a userspace daemon that
    16→//! materialises a virtual `/dev` tree for the guest Android, installs
    17→//! a seccomp filter that traps "dangerous" syscalls, emulates `/proc`
    18→//! and `/sys`, manages the per-VM mount namespace, and exec's the
    19→//! guest's `/system/bin/init`. This crate is that daemon.
    20→//!
    21→//! # Build
    22→//!
    23→//! The crate builds as BOTH a cdylib (`libkr64.so` — directly
    24→//! executable via the PIE hack) AND a regular binary (`kr64`). See
    25→//! `Cargo.toml` and `build.rs` for details.
    26→//!
    27→//! # Usage
    28→//!
    29→//! ```sh
    30→//! # Direct execution via the PIE-hacked cdylib:
    31→//! ./libkr64.so --rootfs /data/.../fs --data-dir /data/... --vmid 0

    33→//! # Or via the regular binary:
    34→//! ./kr64 --rootfs /data/.../fs --data-dir /data/... --vmid 0

   283→/// Run the daemon. Returns the exit code.
   284→///
   285→/// This is the shared entry point called by both:
   286→///   * `kr64_main` (the cdylib's PIE entry point, see below), and
   287→///   * `main()` in `src/main.rs` (the bin target).
   288→///
   289→/// Steps:
   290→///   1. Parse args.
   291→///   2. Create all virtual `/dev` devices (qemu_pipe, touch, key,
   292→///      event, gb, gb2).
   293→///   3. Populate `/proc` (version, cpuinfo, meminfo, cmdline, …).
   294→///   4. Set up mount namespace (unshare + bind mounts + tmpfs mounts).
   295→///   5. Fork:
   296→///      - Child: pivot_root → install seccomp → exec /system/bin/init.
   297→///      - Parent: run the device-accept loop (spawns one thread per
   298→///        device socket; for the MVP each thread just accepts and
   299→///        echoes).
   300→///   6. Wait for the child (the guest init) to exit; propagate its
   301→///      exit code.
   302→pub fn run<I: IntoIterator<Item = String>>(args: I) -> i32 {
   303→    let cfg = match parse_args(args) {
   304→        Ok(c) => c,
   305→        Err(e) => {
   306→            // If --help, print to stdout and exit 0; otherwise stderr
   307→            // and exit 2.
   308→            if e.starts_with("twoyi kr64") {
   309→                println!("{}", e);
   310→                return 0;
   311→            }
   312→            eprintln!("kr64: {}", e);
   313→            return 2;
   314→        }
   315→    };
   316→
   317→    info!("[KR64] starting daemon with config: {:?}", cfg);
   318→
   319→    // ---------------------------------------------------------------
   320→    // Step 2: create virtual /dev devices.
   321→    // ---------------------------------------------------------------
   322→    let device_set = match devices::create_all_devices(&cfg.rootfs, &cfg.data_dir) {
   323→        Ok(d) => d,
   324→        Err(e) => {
   325→            error!("[KR64] failed to create devices: {}", e);
   326→            return 1;
   327→        }
   328→    };
   329→
   330→    // Create the marker files the guest init looks for.
   331→    if let Err(e) = devices::create_coldboot_done_marker(&cfg.rootfs) {
   332→        warning!("[KR64] failed to create .coldboot_done marker: {}", e);
   333→    }
   334→    if let Err(e) = devices::create_busybox_marker(&cfg.rootfs) {
   335→        warning!("[KR64] failed to create .busybox marker: {}", e);
   336→    }
   337→
   338→    // ---------------------------------------------------------------
   339→    // Step 2.5: create the per-VM binder device + spawn the binder
   340→    // proxy. See `download/BINDER_SKELETON.md` and
   341→    // `app/rs/kr64/src/binder.rs` for the full story. The short
   342→    // version: this creates `{rootfs}/vm{id}/dev/binder` as a Unix
   343→    // socket + a `{rootfs}/dev/binder` symlink, and spawns an
   344→    // accept-thread + worker-pool that dispatches BINDER_* ioctls
   345→    // from the guest. The handle is held for the lifetime of `run()`
   346→    // so the proxy is shut down when the guest exits.
   347→    // ---------------------------------------------------------------
   348→    let _binder_handle = match binder::create_binder_device(&cfg.rootfs, cfg.vmid)
   349→        .and_then(|path| binder::BinderProxy::new(cfg.vmid, &path))
   350→        .and_then(|proxy| proxy.spawn())
   351→    {
   352→        Ok(h) => {
   353→            info!("[KR64] binder proxy listening at {} (vm{})", h.path(), cfg.vmid);
   354→            Some(h)
   355→        }
   356→        Err(e) => {
   357→            // Non-fatal: the guest can still boot against the host's
   358→            // /dev/binder if it's bind-mounted in (the current twoyi
   359→            // approach). The binder proxy is only needed for full
   360→            // service-manager virtualisation.
   361→            warning!("[KR64] failed to start binder proxy: {} — falling back to host binder", e);
   362→            None
   363→        }
   364→    };
   365→
   366→    // ---------------------------------------------------------------
   367→    // Step 3: populate /proc.
   368→    // ---------------------------------------------------------------
   369→    // We synthesise for an 8-core, 4 GB guest by default. A production
   370→    // version would query the host's CPU count and memory size and
   371→    // scale these accordingly.
   372→    let cpu_count = 8u32;
   373→    let mem_mb = 4096u64;
   374→    if let Err(e) = proc_emu::populate_proc(&cfg.rootfs, cpu_count, mem_mb) {
   375→        warning!("[KR64] failed to populate /proc: {}", e);
   376→    }
   377→
   378→    // ---------------------------------------------------------------
   379→    // Step 4: set up mount namespace + bind mounts + tmpfs.
   380→    // ---------------------------------------------------------------
   381→    // NOTE: We do NOT do this yet — the mount setup has to happen
   382→    // inside the forked child, because once we pivot_root we lose
   383→    // access to the host's paths (and the parent needs to keep
   384→    // accepting connections on the device sockets it bound above).
   385→    //
   386→    // The flow is:
   387→    //   parent: bind device sockets (already done above)
   388→    //   parent: fork()
   389→    //     child: mount_mgr::setup_mounts(cfg)
   390→    //     child: seccomp::install()
   391→    //     child: execve(/system/bin/init)
   392→    //   parent: accept loop on device sockets
   393→
   394→    // ---------------------------------------------------------------
   395→    // Step 5: fork + exec the guest.
   396→    // ---------------------------------------------------------------
   397→    info!("[KR64] forking guest process");
   398→    let pid = unsafe { libc::fork() };
   399→    if pid < 0 {
   400→        let e = std::io::Error::last_os_error();
   401→        error!("[KR64] fork failed: {}", e);
   402→        return 1;
   403→    }
   404→
   405→    if pid == 0 {
   406→        // ----- CHILD (will become the guest init) -----
   407→        info!("[KR64][child] setting up mounts");
   408→        let mount_cfg = mount_mgr::MountConfig {
   409→            rootfs:           cfg.rootfs.clone(),
   410→            rom_dir:          cfg.rom_dir.clone(),
   411→            use_namespaces:   cfg.use_namespaces,
   412→            read_only_rom:    cfg.read_only_rom,
   413→        };
   414→        if let Err(e) = mount_mgr::setup_mounts(&mount_cfg) {
   415→            error!("[KR64][child] mount setup failed: {}", e);
   416→            unsafe { libc::_exit(1) };
   417→        }
   418→
   419→        // Install seccomp filter on the guest process. This MUST be
   420→        // done after mount setup (otherwise the mount syscalls would
   421→        // be trapped and we couldn't set up the rootfs).
   422→        if cfg.install_seccomp {
   423→            info!("[KR64][child] installing seccomp filter");
   424→            if let Err(e) = seccomp::install() {
   425→                error!("[KR64][child] seccomp install failed: {}", e);
   426→                // Continue anyway — the guest will see the host's
   427→                // unfiltered syscalls, which is insecure but lets
   428→                // debugging proceed.
   429→            }
   430→        }
   431→
   432→        // Exec the guest init.
   433→        info!("[KR64][child] exec'ing guest init at {}", cfg.init_path);
   434→        let init_cstr = CString::new(cfg.init_path.as_str()).unwrap();
   435→        let argv0 = CString::new(cfg.init_path.as_str()).unwrap();
   436→        let argv: [*const libc::c_char; 2] = [argv0.as_ptr(), std::ptr::null()];
   437→        let envp: [*const libc::c_char; 1] = [std::ptr::null()];
   438→
   439→        // Set a few environment variables the guest init may look for.
   440→        // (execve replaces envp entirely, so we have to construct it.)
   441→        // For the MVP we just pass an empty envp — the guest's init
   442→        // builds its own environment from /proc/cmdline + properties.
   443→        let r = unsafe { libc::execve(init_cstr.as_ptr(), argv.as_ptr(), envp.as_ptr()) };
   444→        let e = std::io::Error::last_os_error();
   445→        error!("[KR64][child] execve({}) failed (r={}): {}", cfg.init_path, r, e);
   446→        unsafe { libc::_exit(127) };
   447→    }
   448→
   449→    // ----- PARENT (the daemon) -----
   450→    info!("[KR64][parent] guest pid = {}", pid);
   451→
   452→    // Spawn one accept thread per device socket. For the MVP each
   453→    // thread just accepts connections and immediately closes them
   454→    // (echoing a single byte so the guest sees SOME response). The
   455→    // production version will dispatch to per-device handlers:
   456→    //   qemu_pipe → openglrenderer::pipe
   457→    //   touch     → input::touch_server
   458→    //   key       → input::key_server
   459→    //   event     → TwoyiSocketServer (event IPC)
   460→    //   gb/gb2    → openglrenderer::gralloc
   461→    spawn_accept_thread(device_set.qemu_pipe, "qemu_pipe");
   462→    spawn_accept_thread(device_set.touch,     "touch");
   463→    spawn_accept_thread(device_set.key,       "key");
   464→    spawn_accept_thread(device_set.event,     "event");
   465→    spawn_accept_thread(device_set.gb.gb,     "gb");
   466→    spawn_accept_thread(device_set.gb.gb2,    "gb2");
   467→
   468→    // ---------------------------------------------------------------
   469→    // Step 6: wait for the guest to exit.
   470→    // ---------------------------------------------------------------
   471→    let mut status: libc::c_int = 0;
   472→    loop {
   473→        let waited = unsafe { libc::waitpid(pid, &mut status, 0) };
   474→        if waited == -1 {
   475→            let e = std::io::Error::last_os_error();
   476→            if e.raw_os_error() == Some(libc::EINTR) {
   477→                continue;
   478→            }
   479→            error!("[KR64][parent] waitpid failed: {}", e);
   480→            return 1;
   481→        }
   482→        break;
   483→    }
   484→
   485→    if libc::WIFEXITED(status) {
   486→        let code = libc::WEXITSTATUS(status);
   487→        info!("[KR64][parent] guest exited with status {}", code);
   488→        return code;
   489→    }
   490→    if libc::WIFSIGNALED(status) {
   491→        let sig = libc::WTERMSIG(status);
   492→        warning!("[KR64][parent] guest killed by signal {}", sig);
   493→        return 128 + sig;
   494→    }
   495→
   496→    warning!("[KR64][parent] guest waitpid returned unexpected status {}", status);
   497→    1
   498→}

   501→/// and immediately closes them (MVP placeholder).
   502→///
   503→/// The thread takes ownership of the underlying `UnixListener` (so
   504→/// the parent `DeviceSocket` is consumed). For the production version
   505→/// this will be replaced with per-device handler dispatch.
   506→fn spawn_accept_thread(mut dev: devices::DeviceSocket, name: &'static str) {
   507→    let listener = match dev.take_listener() {
   508→        Some(l) => l,
   509→        None => {
   510→            warning!("[KR64] cannot spawn accept thread for {}: listener already taken", name);
   511→            return;
   512→        }
   513→    };
   514→    // Make the listening socket non-blocking so the thread can poll
   515→    // for shutdown signals (in the MVP we just loop forever).
   516→    let fd = listener.as_raw_fd();
   517→    let _ = unsafe { libc::fcntl(fd, libc::F_SETFL, libc::O_NONBLOCK) };
   518→
   519→    std::thread::Builder::new()
   520→        .name(format!("kr64-accept-{}", name))
   521→        .spawn(move || {
   522→            info!("[KR64][{}] accept thread started (fd={})", name, fd);
   523→            loop {
   524→                match listener.accept() {
   525→                    Ok((mut stream, _addr)) => {
   526→                        info!("[KR64][{}] client connected", name);
   527→                        // Echo a single byte so the guest sees SOME response.
   528→                        // (Many of the device protocols expect a handshake
   529→                        // byte — e.g. the touch device sends a device_info
   530→                        // struct on connect, which the guest reads before
   531→                        // sending anything. The production version will
   532→                        // dispatch to the right handler.)
   533→                        use std::io::Write;
   534→                        let _ = stream.write_all(&[0u8]);
   535→                    }
   536→                    Err(e) if e.kind() == std::io::ErrorKind::WouldBlock => {
   537→                        // No pending connection — sleep briefly to avoid
   538→                        // spinning. (Real implementation would use epoll.)
   539→                        std::thread::sleep(std::time::Duration::from_millis(50));
   540→                    }
   541→                    Err(e) => {
   542→                        warning!("[KR64][{}] accept error: {}", name, e);
   543→                        std::thread::sleep(std::time::Duration::from_millis(100));
   544→                    }
   545→                }
   546→            }
   547→        })
   548→        .expect("spawn kr64 accept thread");
   549→}

   551→// ============================================================================
   552→// cdylib entry point — used when `libkr64.so` is exec'd directly via
   553→// the PIE hack. The ELF entry point is set to `kr64_main` via the
   554→// `-Wl,-e,kr64_main` link flag in build.rs / .cargo/config.toml.
   555→// ============================================================================
   556→
   557→/// Entry point for the cdylib (libkr64.so). Mirrors the C `main`
   558→/// signature `(argc, argv) → int` so the standard C runtime can call
   559→/// it.
   560→///
   561→/// When the kernel exec's `libkr64.so`, the dynamic linker loads it,
   562→/// runs `.init_array`, and jumps to `kr64_main` (because we set
   563→/// `-Wl,-e,kr64_main`). The args come from the kernel's stack as
   564→/// `argc` + `argv[]`, just like a regular C `main`.
   565→#[no_mangle]
   566→pub extern "C" fn kr64_main(argc: libc::c_int, argv: *const *const libc::c_char) -> libc::c_int {
   567→    // Convert C argv → Rust Vec<String>.
   568→    let args: Vec<String> = if argc <= 0 || argv.is_null() {
   569→        Vec::new()
   570→    } else {
   571→        (0..argc as isize)
   572→            .map(|i| unsafe {
   573→                let p = *argv.offset(i);
   574→                if p.is_null() { String::new() }
   575→                else { std::ffi::CStr::from_ptr(p).to_string_lossy().to_string() }
   576→            })
   577→            .collect()
   578→    };
   579→
   580→    run(args)
   581→}
   582→
   583→// Reference the interp symbol from interp.c (built by build.rs) so the
   584→// linker keeps the `.interp` section in the final libkr64.so. Without
   585→// this reference, the linker may GC the `.interp` section as unused,
   586→// and the resulting .so won't be directly executable.
   587→extern "C" {
   588→    #[link_name = "interp"]
   589→    static INTERP: [u8; 0];
   590→}
   591→
   592→#[used]
   593→static INTERP_REF: &'static [u8; 0] = unsafe { &INTERP };
