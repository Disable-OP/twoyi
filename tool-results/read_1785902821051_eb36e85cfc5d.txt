    12→//! Twoyi kernel-replacement daemon — the Rust port of VM's `libkr64.so`.
    13→//!
    14→//! See `VM_KR64_ANALYSIS.md` and `GSI_BOOT_PLAN.md` for the full
    15→//! background. The short version: twoyi needs a userspace daemon that
    16→//! materialises a virtual `/dev` tree for the guest Android, installs
    17→//! a seccomp filter that traps "dangerous" syscalls, emulates `/proc`
    18→//! and `/sys`, manages the per-VM mount namespace, and exec's the
    19→//! guest's `/system/bin/init`. This crate is that daemon.
    20→//!
    21→//! # Build
    22→//!
    23→//! The crate builds as BOTH a cdylib (`libkr64.so` — directly
    24→//! executable via the PIE hack) AND a regular binary (`kr64`). See
    25→//! `Cargo.toml` and `build.rs` for details.
    26→//!
    27→//! # Usage
    28→//!
    29→//! ```sh
    30→//! # Direct execution via the PIE-hacked cdylib:
    31→//! ./libkr64.so --rootfs /data/.../fs --data-dir /data/... --vmid 0
    32→//!
    33→//! # Or via the regular binary:
    34→//! ./kr64 --rootfs /data/.../fs --data-dir /data/... --vmid 0
    35→//! ```
    36→//!
    37→//! # Module layout
    38→//!
    39→//! * [`devices`]    — virtual `/dev` device creation (qemu_pipe,
    40→//!                    touch, key, event, gb, gb2).
    41→//! * [`binder`]     — per-VM `/vm{id}/dev/binder` Unix socket + binder
    42→//!                    transaction proxy (skeleton; see
    43→//!                    `download/BINDER_SKELETON.md`).
    44→//! * [`audio`]      — virtual `/dev/audio` Unix socket + bidirectional
    45→//!                    PCM pump (playback + capture skeleton; see
    46→//!                    `download/AUDIO_SENSOR_HAL.md`).
    47→//! * [`seccomp`]    — BPF seccomp filter + SIGSYS handler.
    48→//! * [`proc_emu`]   — synthesised `/proc` tree (version, cpuinfo,
    49→//!                    meminfo, cmdline, self/, sys/).
    50→//! * [`mount_mgr`]  — `unshare(CLONE_NEWNS)` + `pivot_root` + tmpfs
    51→//!                    mounts for /dev, /proc, /sys, /system, /vendor.

   287→/// Run the daemon. Returns the exit code.
   288→///
   289→/// This is the shared entry point called by both:
   290→///   * `kr64_main` (the cdylib's PIE entry point, see below), and
   291→///   * `main()` in `src/main.rs` (the bin target).
   292→///
   293→/// Steps:
   294→///   1. Parse args.
   295→///   2. Create all virtual `/dev` devices (qemu_pipe, touch, key,
   296→///      event, gb, gb2).
   297→///   3. Populate `/proc` (version, cpuinfo, meminfo, cmdline, …).
   298→///   4. Set up mount namespace (unshare + bind mounts + tmpfs mounts).
   299→///   5. Fork:
   300→///      - Child: pivot_root → install seccomp → exec /system/bin/init.
   301→///      - Parent: run the device-accept loop (spawns one thread per
   302→///        device socket; for the MVP each thread just accepts and
   303→///        echoes).
   304→///   6. Wait for the child (the guest init) to exit; propagate its
   305→///      exit code.
   306→pub fn run<I: IntoIterator<Item = String>>(args: I) -> i32 {
   307→    let cfg = match parse_args(args) {
   308→        Ok(c) => c,
   309→        Err(e) => {
   310→            // If --help, print to stdout and exit 0; otherwise stderr
   311→            // and exit 2.
   312→            if e.starts_with("twoyi kr64") {
   313→                println!("{}", e);
   314→                return 0;
   315→            }
   316→            eprintln!("kr64: {}", e);
   317→            return 2;
   318→        }
   319→    };
   320→
   321→    info!("[KR64] starting daemon with config: {:?}", cfg);
   322→
   323→    // ---------------------------------------------------------------
   324→    // Step 2: create virtual /dev devices.
   325→    // ---------------------------------------------------------------
   326→    let device_set = match devices::create_all_devices(&cfg.rootfs, &cfg.data_dir) {
   327→        Ok(d) => d,
   328→        Err(e) => {
   329→            error!("[KR64] failed to create devices: {}", e);
   330→            return 1;
   331→        }
   332→    };
   333→
   334→    // Create the marker files the guest init looks for.
   335→    if let Err(e) = devices::create_coldboot_done_marker(&cfg.rootfs) {
   336→        warning!("[KR64] failed to create .coldboot_done marker: {}", e);
   337→    }
   338→    if let Err(e) = devices::create_busybox_marker(&cfg.rootfs) {
   339→        warning!("[KR64] failed to create .busybox marker: {}", e);
   340→    }
   341→
   342→    // ---------------------------------------------------------------
   343→    // Step 2.5: create the per-VM binder device + spawn the binder
   344→    // proxy. See `download/BINDER_SKELETON.md` and
   345→    // `app/rs/kr64/src/binder.rs` for the full story. The short
   346→    // version: this creates `{rootfs}/vm{id}/dev/binder` as a Unix
   347→    // socket + a `{rootfs}/dev/binder` symlink, and spawns an
   348→    // accept-thread + worker-pool that dispatches BINDER_* ioctls
   349→    // from the guest. The handle is held for the lifetime of `run()`
   350→    // so the proxy is shut down when the guest exits.
   351→    // ---------------------------------------------------------------
   352→    let _binder_handle = match binder::create_binder_device(&cfg.rootfs, cfg.vmid)
   353→        .and_then(|path| binder::BinderProxy::new(cfg.vmid, &path))
   354→        .and_then(|proxy| proxy.spawn())
   355→    {
   356→        Ok(h) => {
   357→            info!("[KR64] binder proxy listening at {} (vm{})", h.path(), cfg.vmid);
   358→            Some(h)
   359→        }
   360→        Err(e) => {
   361→            // Non-fatal: the guest can still boot against the host's
   362→            // /dev/binder if it's bind-mounted in (the current twoyi
   363→            // approach). The binder proxy is only needed for full
   364→            // service-manager virtualisation.
   365→            warning!("[KR64] failed to start binder proxy: {} — falling back to host binder", e);
   366→            None
   367→        }
   368→    };
   369→
   370→    // ---------------------------------------------------------------
   371→    // Step 2.6: create the audio device + spawn the accept/pump
   372→    // thread. See `download/AUDIO_SENSOR_HAL.md` and
   373→    // `app/rs/kr64/src/audio.rs` for the full story. The short
   374→    // version: this creates `{rootfs}/dev/audio` as a Unix socket
   375→    // and spawns an accept-thread + worker-pool that reads a
   376→    // 16-byte header per connection and pumps raw PCM in both
   377→    // directions. The actual AudioTrack/AudioRecord integration is
   378→    // stubbed (no JNI yet) — the pump compiles and exercises the
   379→    // protocol but produces no sound until the Java side is wired
   380→    // up in a follow-up task.
   381→    // ---------------------------------------------------------------
   382→    let _audio_handle = match audio::create_audio_device(&cfg.rootfs)
   383→        .and_then(|dev| dev.spawn())
   384→    {
   385→        Ok(h) => {
   386→            info!("[KR64] audio device listening at {}", h.path());
   387→            Some(h)
   388→        }
   389→        Err(e) => {
   390→            // Non-fatal: the guest can still boot without sound —
   391→            // AudioFlinger's connect() to /dev/audio will fail and
   392→            // the guest's audio HAL will fall back to silence / a
   393→            // null output. Sound is the user's primary use case
   394→            // though, so this warning is worth surfacing.
   395→            warning!("[KR64] failed to start audio device: {} — guest will have no sound", e);
   396→            None
   397→        }
   398→    };
   399→
   400→    // ---------------------------------------------------------------
   401→    // Step 3: populate /proc.
   402→    // ---------------------------------------------------------------
   403→    // We synthesise for an 8-core, 4 GB guest by default. A production
   404→    // version would query the host's CPU count and memory size and
   405→    // scale these accordingly.
   406→    let cpu_count = 8u32;
   407→    let mem_mb = 4096u64;
   408→    if let Err(e) = proc_emu::populate_proc(&cfg.rootfs, cpu_count, mem_mb) {
   409→        warning!("[KR64] failed to populate /proc: {}", e);
   410→    }
   411→
   412→    // ---------------------------------------------------------------
   413→    // Step 4: set up mount namespace + bind mounts + tmpfs.
   414→    // ---------------------------------------------------------------
   415→    // NOTE: We do NOT do this yet — the mount setup has to happen
   416→    // inside the forked child, because once we pivot_root we lose
   417→    // access to the host's paths (and the parent needs to keep
   418→    // accepting connections on the device sockets it bound above).
   419→    //
   420→    // The flow is:
   421→    //   parent: bind device sockets (already done above)
   422→    //   parent: fork()
   423→    //     child: mount_mgr::setup_mounts(cfg)
   424→    //     child: seccomp::install()
   425→    //     child: execve(/system/bin/init)
   426→    //   parent: accept loop on device sockets
   427→
   428→    // ---------------------------------------------------------------
   429→    // Step 5: fork + exec the guest.
   430→    // ---------------------------------------------------------------
   431→    info!("[KR64] forking guest process");
   432→    let pid = unsafe { libc::fork() };
   433→    if pid < 0 {
   434→        let e = std::io::Error::last_os_error();
   435→        error!("[KR64] fork failed: {}", e);
   436→        return 1;
   437→    }
   438→
   439→    if pid == 0 {
   440→        // ----- CHILD (will become the guest init) -----
   441→        info!("[KR64][child] setting up mounts");
   442→        let mount_cfg = mount_mgr::MountConfig {
   443→            rootfs:           cfg.rootfs.clone(),
   444→            rom_dir:          cfg.rom_dir.clone(),
   445→            use_namespaces:   cfg.use_namespaces,
   446→            read_only_rom:    cfg.read_only_rom,
   447→        };
   448→        if let Err(e) = mount_mgr::setup_mounts(&mount_cfg) {
   449→            error!("[KR64][child] mount setup failed: {}", e);
   450→            unsafe { libc::_exit(1) };
   451→        }
   452→
   453→        // Install seccomp filter on the guest process. This MUST be
   454→        // done after mount setup (otherwise the mount syscalls would
   455→        // be trapped and we couldn't set up the rootfs).
   456→        if cfg.install_seccomp {
   457→            info!("[KR64][child] installing seccomp filter");
   458→            if let Err(e) = seccomp::install() {
   459→                error!("[KR64][child] seccomp install failed: {}", e);
   460→                // Continue anyway — the guest will see the host's
   461→                // unfiltered syscalls, which is insecure but lets
   462→                // debugging proceed.
   463→            }
   464→        }
   465→
   466→        // Exec the guest init.
   467→        info!("[KR64][child] exec'ing guest init at {}", cfg.init_path);
   468→        let init_cstr = CString::new(cfg.init_path.as_str()).unwrap();
   469→        let argv0 = CString::new(cfg.init_path.as_str()).unwrap();
   470→        let argv: [*const libc::c_char; 2] = [argv0.as_ptr(), std::ptr::null()];
   471→        let envp: [*const libc::c_char; 1] = [std::ptr::null()];
   472→
   473→        // Set a few environment variables the guest init may look for.
   474→        // (execve replaces envp entirely, so we have to construct it.)
   475→        // For the MVP we just pass an empty envp — the guest's init
   476→        // builds its own environment from /proc/cmdline + properties.
   477→        let r = unsafe { libc::execve(init_cstr.as_ptr(), argv.as_ptr(), envp.as_ptr()) };
   478→        let e = std::io::Error::last_os_error();
   479→        error!("[KR64][child] execve({}) failed (r={}): {}", cfg.init_path, r, e);
   480→        unsafe { libc::_exit(127) };
   481→    }
   482→
   483→    // ----- PARENT (the daemon) -----
   484→    info!("[KR64][parent] guest pid = {}", pid);
   485→
   486→    // Spawn one accept thread per device socket. For the MVP each
   487→    // thread just accepts connections and immediately closes them
   488→    // (echoing a single byte so the guest sees SOME response). The
   489→    // production version will dispatch to per-device handlers:
   490→    //   qemu_pipe → openglrenderer::pipe
   491→    //   touch     → input::touch_server
   492→    //   key       → input::key_server
   493→    //   event     → TwoyiSocketServer (event IPC)
   494→    //   gb/gb2    → openglrenderer::gralloc
   495→    spawn_accept_thread(device_set.qemu_pipe, "qemu_pipe");
   496→    spawn_accept_thread(device_set.touch,     "touch");
   497→    spawn_accept_thread(device_set.key,       "key");
   498→    spawn_accept_thread(device_set.event,     "event");
   499→    spawn_accept_thread(device_set.gb.gb,     "gb");
   500→    spawn_accept_thread(device_set.gb.gb2,    "gb2");
   501→
   502→    // ---------------------------------------------------------------
   503→    // Step 6: wait for the guest to exit.
   504→    // ---------------------------------------------------------------
   505→    let mut status: libc::c_int = 0;
   506→    loop {
   507→        let waited = unsafe { libc::waitpid(pid, &mut status, 0) };
   508→        if waited == -1 {
   509→            let e = std::io::Error::last_os_error();
   510→            if e.raw_os_error() == Some(libc::EINTR) {
   511→                continue;
   512→            }
   513→            error!("[KR64][parent] waitpid failed: {}", e);
   514→            return 1;
   515→        }
   516→        break;
   517→    }
   518→
   519→    if libc::WIFEXITED(status) {
   520→        let code = libc::WEXITSTATUS(status);
   521→        info!("[KR64][parent] guest exited with status {}", code);
   522→        return code;
   523→    }
   524→    if libc::WIFSIGNALED(status) {
   525→        let sig = libc::WTERMSIG(status);
   526→        warning!("[KR64][parent] guest killed by signal {}", sig);
   527→        return 128 + sig;
   528→    }
   529→
   530→    warning!("[KR64][parent] guest waitpid returned unexpected status {}", status);
   531→    1
   532→}

   534→/// Spawn a thread that accepts connections on the given device socket
   535→/// and immediately closes them (MVP placeholder).
   536→///
   537→/// The thread takes ownership of the underlying `UnixListener` (so
   538→/// the parent `DeviceSocket` is consumed). For the production version
   539→/// this will be replaced with per-device handler dispatch.
   540→fn spawn_accept_thread(mut dev: devices::DeviceSocket, name: &'static str) {
   541→    let listener = match dev.take_listener() {
   542→        Some(l) => l,
   543→        None => {
   544→            warning!("[KR64] cannot spawn accept thread for {}: listener already taken", name);
   545→            return;
   546→        }
   547→    };
   548→    // Make the listening socket non-blocking so the thread can poll
   549→    // for shutdown signals (in the MVP we just loop forever).
   550→    let fd = listener.as_raw_fd();
   551→    let _ = unsafe { libc::fcntl(fd, libc::F_SETFL, libc::O_NONBLOCK) };
   552→
   553→    std::thread::Builder::new()
   554→        .name(format!("kr64-accept-{}", name))
   555→        .spawn(move || {
   556→            info!("[KR64][{}] accept thread started (fd={})", name, fd);
   557→            loop {
   558→                match listener.accept() {
   559→                    Ok((mut stream, _addr)) => {
   560→                        info!("[KR64][{}] client connected", name);
   561→                        // Echo a single byte so the guest sees SOME response.
   562→                        // (Many of the device protocols expect a handshake
   563→                        // byte — e.g. the touch device sends a device_info
   564→                        // struct on connect, which the guest reads before
   565→                        // sending anything. The production version will
   566→                        // dispatch to the right handler.)
   567→                        use std::io::Write;
   568→                        let _ = stream.write_all(&[0u8]);
   569→                    }
   570→                    Err(e) if e.kind() == std::io::ErrorKind::WouldBlock => {
   571→                        // No pending connection — sleep briefly to avoid
   572→                        // spinning. (Real implementation would use epoll.)
   573→                        std::thread::sleep(std::time::Duration::from_millis(50));
   574→                    }
   575→                    Err(e) => {
   576→                        warning!("[KR64][{}] accept error: {}", name, e);
   577→                        std::thread::sleep(std::time::Duration::from_millis(100));
   578→                    }
   579→                }
   580→            }
   581→        })
   582→        .expect("spawn kr64 accept thread");
   583→}

   585→// ============================================================================
   586→// cdylib entry point — used when `libkr64.so` is exec'd directly via
   587→// the PIE hack. The ELF entry point is set to `kr64_main` via the
   588→// `-Wl,-e,kr64_main` link flag in build.rs / .cargo/config.toml.
   589→// ============================================================================
   590→
   591→/// Entry point for the cdylib (libkr64.so). Mirrors the C `main`
   592→/// signature `(argc, argv) → int` so the standard C runtime can call
   593→/// it.
   594→///
   595→/// When the kernel exec's `libkr64.so`, the dynamic linker loads it,
   596→/// runs `.init_array`, and jumps to `kr64_main` (because we set
   597→/// `-Wl,-e,kr64_main`). The args come from the kernel's stack as
   598→/// `argc` + `argv[]`, just like a regular C `main`.
   599→#[no_mangle]
   600→pub extern "C" fn kr64_main(argc: libc::c_int, argv: *const *const libc::c_char) -> libc::c_int {
   601→    // Convert C argv → Rust Vec<String>.
   602→    let args: Vec<String> = if argc <= 0 || argv.is_null() {
   603→        Vec::new()
   604→    } else {
   605→        (0..argc as isize)
   606→            .map(|i| unsafe {
   607→                let p = *argv.offset(i);
   608→                if p.is_null() { String::new() }
   609→                else { std::ffi::CStr::from_ptr(p).to_string_lossy().to_string() }
   610→            })
   611→            .collect()
   612→    };
   613→
   614→    run(args)
   615→}
   616→
   617→// Reference the interp symbol from interp.c (built by build.rs) so the
   618→// linker keeps the `.interp` section in the final libkr64.so. Without
   619→// this reference, the linker may GC the `.interp` section as unused,
   620→// and the resulting .so won't be directly executable.
   621→extern "C" {
   622→    #[link_name = "interp"]
   623→    static INTERP: [u8; 0];
   624→}
   625→
   626→#[used]
   627→static INTERP_REF: &'static [u8; 0] = unsafe { &INTERP };
