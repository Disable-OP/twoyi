    85→    public static void ensureBootFiles(Context context) {
    86→
    87→        // Kill orphan container processes FIRST so they don't interfere with
    88→        // directory setup or hold on to stale dalvik-cache entries.
    89→        killOrphanProcess();
    90→
    91→        // Ensure /data/local/tmp exists with world-writable permissions.
    92→        // twoyi's init.rc omits the mkdir for /data/local/tmp that AOSP includes,
    93→        // so adbd never has a place to push APKs during `adb install`.
    94→        // This is done here (not via `adb shell mkdir` in the Installer) to avoid
    95→        // a synchronous adb call that hangs when adbd is unresponsive.
    96→        // chmod 777 ensures adbd can write regardless of which UID it runs as.
    97→        ensureDataLocalTmp(context);
    98→
    99→        // Patch services.jar to fix the in-container APK install failure.
   100→        // Android 8.0 (SDK 26) PackageInstallerSession.openWriteInternal() calls
   101→        // target.delete() then Os.stat(target) when offsetBytes==0.  Because the
   102→        // file never existed, Os.stat() throws ENOENT.  The SDK 26 code path does
   103→        // not handle ENOENT, causing "stat failed: ENOENT" on every in-container
   104→        // install attempt.  The fix replaces the throw with "stat = null", matching
   105→        // the SDK 27 behaviour that was added in AOSP to handle this exact case.
   106→        patchServicesJarForPackageInstaller(context);
   107→
   108→        // <rootdir>/dev/
   109→        File devDir = new File(getRootfsDir(context), "dev");
   110→        ensureDir(new File(devDir, "input"));
   111→        ensureDir(new File(devDir, "socket"));
   112→        ensureDir(new File(devDir, "maps"));
   113→
   114→        ensureDir(new File(context.getDataDir(), "socket"));
   115→
   116→        createLoaderSymlink(context);
   117→
   118→        saveLastKmsg(context);
   119→    }

   265→    private static final String PREF_HOST_FINGERPRINT = "host_build_fingerprint";
   266→
   267→    /**
   268→     * Clears the guest Android's dalvik-cache only when the host build fingerprint
   269→     * has changed since the last successful clear (i.e. after a host OTA update).
   270→     *
   271→     * <p>Background: the container's ART OAT/VDEX files in dalvik-cache are compiled
   272→     * against the host ART version.  When the host receives an OTA update the ART
   273→     * version changes, making all existing OAT entries stale.  If the twoyi process
   274→     * has been alive across the OTA (e.g. Xiaomi "frozen" app resurrection), calling
   275→     * clearDalvikCache only in attachBaseContext is not enough because that runs just
   276→     * once per process lifetime.  This method is called synchronously on the UI thread
   277→     * in bootSystem() before addView(mSurfaceView), so the cache is guaranteed to be
   278→     * fully cleared before Renderer.init() starts the container.
   279→     */
   280→    public static void clearDalvikCacheIfNeeded(Context context) {
   281→        String currentFingerprint = Build.FINGERPRINT;
   282→        String lastFingerprint = AppKV.getStringConfig(context, PREF_HOST_FINGERPRINT, "");
   283→        if (!currentFingerprint.equals(lastFingerprint)) {
   284→            Log.i(TAG, "Host fingerprint changed (" + lastFingerprint + " → " + currentFingerprint + "), clearing dalvik-cache");
   285→            clearDalvikCache(context);
   286→            AppKV.setStringConfig(context, PREF_HOST_FINGERPRINT, currentFingerprint);
   287→        } else {
   288→            Log.i(TAG, "Host fingerprint unchanged, skipping dalvik-cache clear");
   289→        }
   290→    }

   338→    // =========================================================================
   339→    // services.jar DEX patch – fix PackageInstallerSession.openWriteInternal()
   340→    // =========================================================================
   341→    //
   342→    // Root cause (Android 8.0 / SDK 26):
   343→    //   openWriteInternal() calls target.delete() then Os.stat(target).
   344→    //   When offsetBytes == 0 (a fresh install) the file was just deleted (or
   345→    //   never existed), so Os.stat() throws ErrnoException(ENOENT).  SDK 26
   346→    //   code does NOT handle ENOENT and re-throws as
   347→    //     IOException("stat failed: ENOENT (No such file or directory)")
   348→    //   making every GUI-triggered in-container APK install fail immediately.
   349→    //
   350→    // Fix (mirrors the SDK 27 change in AOSP):
   351→    //   Replace the two-byte "throw v_ioe" instruction inside the ErrnoException
   352→    //   catch block with "const/4 v_stat, 0" (sets the stat register to null).
   353→    //   The code that follows already handles stat == null by opening the file
   354→    //   with O_CREAT | O_TRUNC, so the install proceeds normally.
   355→    //
   356→    // Encoding equivalence (both instructions are exactly 2 bytes in DEX):
   357→    //   throw  vN  =  0x27  0xNN
   358→    //   const/4 vM, 0  =  0x12  0x0M   (value 0 in the high nibble)
   359→    // =========================================================================
   360→
   361→    /**
   362→     * Patches {@code system/framework/services.jar} inside the container rootfs
   363→     * to fix the ENOENT bug in {@code PackageInstallerSession.openWriteInternal}.
   364→     * The patch is idempotent: a flag file records the last-patched mod-time of
   365→     * {@code services.jar}, so we only re-patch when the JAR has been replaced.
   366→     */
   367→    private static void patchServicesJarForPackageInstaller(Context context) {
   368→        File rootfsDir   = getRootfsDir(context);
   369→        File servicesJar = new File(rootfsDir, "system/framework/services.jar");
   370→        if (!servicesJar.exists()) {
   371→            Log.w(TAG, "patchServicesJar: services.jar not found, skipping");
   372→            return;
   373→        }
   374→
   375→        // Re-use the flag file inside the rootfs data directory (writable by us).
   376→        File patchFlag = new File(rootfsDir, "data/.twoyi_pi_patched");
   377→        if (patchFlag.exists() && patchFlag.lastModified() >= servicesJar.lastModified()) {
   378→            Log.d(TAG, "patchServicesJar: already up-to-date, skipping");
   379→            return;
   380→        }
   381→
   382→        Log.i(TAG, "patchServicesJar: patching " + servicesJar);
   383→        try {
   384→            byte[] dex = extractClassesDexFromJar(servicesJar);
   385→            if (dex == null) {
   386→                Log.e(TAG, "patchServicesJar: failed to extract classes.dex");
   387→                return;
   388→            }
   389→
   390→            int patchedAt = applyOpenWriteInternalEnoentPatch(dex);
   391→            if (patchedAt < 0) {
   392→                // Pattern not found – ROM may already be patched or have a
   393→                // different code layout.  Mark the flag so we don't retry.
   394→                Log.w(TAG, "patchServicesJar: instruction pattern not found – skipping");
   395→                touchFile(patchFlag);
   396→                return;
   397→            }
   398→            Log.i(TAG, "patchServicesJar: patched throw→const/4 at DEX offset 0x"
   399→                    + Integer.toHexString(patchedAt));
   400→
   401→            recomputeDexChecksums(dex);
   402→
   403→            if (!replaceClassesDexInJar(servicesJar, dex)) {
   404→                Log.e(TAG, "patchServicesJar: failed to write back services.jar");
   405→                return;
   406→            }
   407→
   408→            deleteServicesJarOatFiles(rootfsDir);
   409→            touchFile(patchFlag);
   410→            Log.i(TAG, "patchServicesJar: patch applied successfully");
   411→
   412→        } catch (Exception e) {
   413→            Log.e(TAG, "patchServicesJar: unexpected error", e);
   414→        }
   415→    }

   417→    /**
   418→     * Returns the raw bytes of {@code classes.dex} extracted from a JAR/ZIP,
   419→     * or {@code null} on failure.
   420→     */
   421→    private static byte[] extractClassesDexFromJar(File jar) throws IOException {
   422→        try (ZipFile zf = new ZipFile(jar)) {
   423→            ZipEntry entry = zf.getEntry("classes.dex");
   424→            if (entry == null) {
   425→                Log.w(TAG, "extractClassesDex: no classes.dex in " + jar);
   426→                return null;
   427→            }
   428→            try (InputStream in = zf.getInputStream(entry)) {
   429→                ByteArrayOutputStream baos =
   430→                        new ByteArrayOutputStream((int) entry.getSize());
   431→                byte[] buf = new byte[8192];
   432→                int n;
   433→                while ((n = in.read(buf)) >= 0) baos.write(buf, 0, n);
   434→                return baos.toByteArray();
   435→            }
   436→        }
   437→    }

   439→    /**
   440→     * Binary-patches {@code dex} in-place.
   441→     *
   442→     * <p>Algorithm:
   443→     * <ol>
   444→     *   <li>Find string-ID for {@code "stat failed: "} in the DEX string pool.</li>
   445→     *   <li>Find the {@code const-string vR, "stat failed: "} instruction
   446→     *       (opcode 0x1a) in the code section.</li>
   447→     *   <li>Scan backwards to find {@code move-exception} (0x0d) – the entry
   448→     *       point of the {@code ErrnoException} catch handler.</li>
   449→     *   <li>Scan further backwards to find {@code move-result-object v_stat}
   450→     *       (0x0c) – the instruction that captures {@code Os.stat()}'s return
   451→     *       value and whose register we must set to {@code null}.</li>
   452→     *   <li>Scan forwards from the {@code const-string} to find the
   453→     *       {@code throw v_ioe} instruction (0x27).</li>
   454→     *   <li>Replace {@code 0x27 vN} with {@code 0x12 0x0M} where M is the
   455→     *       v_stat register index (must be 0–15 for {@code const/4}).</li>
   456→     * </ol>
   457→     *
   458→     * @return the byte offset of the patched instruction, or -1 if the pattern
   459→     *         was not found or cannot be safely patched.
   460→     */
   461→    private static int applyOpenWriteInternalEnoentPatch(byte[] dex) {
   462→        // ── Step 1: find the string ID for "stat failed: " ────────────────────
   463→        final byte[] TARGET = "stat failed: ".getBytes(StandardCharsets.UTF_8);
   464→        int strIdsSize = readInt32LE(dex, 56);
   465→        int strIdsOff  = readInt32LE(dex, 60);
   466→
   467→        int statFailedId = -1;
   468→        for (int i = 0; i < strIdsSize; i++) {
   469→            int dataOff = readInt32LE(dex, strIdsOff + i * 4);
   470→            if (dataOff < 0 || dataOff + 1 + TARGET.length >= dex.length) continue;
   471→            int charCount = readUleb128(dex, dataOff);
   472→            if (charCount != TARGET.length) continue;
   473→            int headerLen = uleb128Size(charCount);
   474→            boolean match = true;
   475→            for (int j = 0; j < TARGET.length; j++) {
   476→                if (dex[dataOff + headerLen + j] != TARGET[j]) { match = false; break; }
   477→            }
   478→            if (match) { statFailedId = i; break; }
   479→        }
   480→        if (statFailedId < 0) {
   481→            Log.w(TAG, "applyPatch: string 'stat failed: ' not found in DEX");
   482→            return -1;
   483→        }
   484→
   485→        // ── Step 2: find const-string vR, statFailedId  (opcode 0x1a) ─────────
   486→        byte idLo = (byte) (statFailedId & 0xff);
   487→        byte idHi = (byte) ((statFailedId >> 8) & 0xff);
   488→        int csOff = -1;
   489→        // Start searching after the string-IDs table (string data can't hold code).
   490→        int codeSearchStart = strIdsOff + strIdsSize * 4;
   491→        for (int i = codeSearchStart; i < dex.length - 4; i++) {
   492→            if ((dex[i] & 0xff) == 0x1a && dex[i + 2] == idLo && dex[i + 3] == idHi) {
   493→                csOff = i;
   494→                break;
   495→            }
   496→        }
   497→        if (csOff < 0) {
   498→            Log.w(TAG, "applyPatch: const-string for 'stat failed: ' not found");
   499→            return -1;
   500→        }
   501→
   502→        // ── Step 3: find move-exception (0x0d) backwards from csOff ──────────
   503→        //   The ErrnoException catch handler starts with move-exception.
   504→        int meOff = -1;
   505→        // Scan in 2-byte steps (Dalvik code units are 2-byte aligned).
   506→        for (int i = csOff - 2; i >= Math.max(0, csOff - 128); i -= 2) {
   507→            if ((dex[i] & 0xff) == 0x0d) { meOff = i; break; }
   508→        }
   509→        if (meOff < 0) { // fallback: byte-by-byte
   510→            for (int i = csOff - 1; i >= Math.max(0, csOff - 128); i--) {
   511→                if ((dex[i] & 0xff) == 0x0d) { meOff = i; break; }
   512→            }
   513→        }
   514→        if (meOff < 0) {
   515→            Log.w(TAG, "applyPatch: move-exception not found before 'stat failed: '");
   516→            return -1;
   517→        }
   518→
   519→        // ── Step 4: find move-result-object (0x0c) backwards from move-exception
   520→        //   This is the instruction that stores Os.stat()'s return value into
   521→        //   v_stat.  It is the last move-result-object before the catch handler.
   522→        int statReg = -1;
   523→        for (int i = meOff - 2; i >= Math.max(0, meOff - 512); i -= 2) {
   524→            if ((dex[i] & 0xff) == 0x0c) { statReg = dex[i + 1] & 0xff; break; }
   525→        }
   526→        if (statReg < 0) { // fallback
   527→            for (int i = meOff - 1; i >= Math.max(0, meOff - 512); i--) {
   528→                if ((dex[i] & 0xff) == 0x0c) { statReg = dex[i + 1] & 0xff; break; }
   529→            }
   530→        }
   531→        if (statReg < 0) {
   532→            Log.w(TAG, "applyPatch: move-result-object for v_stat not found");
   533→            return -1;
   534→        }
   535→        if (statReg > 15) {
   536→            // const/4 can only address registers v0–v15.
   537→            Log.w(TAG, "applyPatch: v_stat = v" + statReg + " is out of range for const/4");
   538→            return -1;
   539→        }
   540→
   541→        // ── Step 5: find throw (0x27) forwards from csOff ────────────────────
   542→        //   After const-string comes string-building code (new-instance,
   543→        //   invoke-virtual, move-result-object …) then invoke-direct for
   544→        //   IOException.<init> and finally throw v_ioe.
   545→        int throwOff = -1;
   546→        int limit = Math.min(csOff + 256, dex.length - 2);
   547→        for (int i = csOff + 4; i < limit; i++) {
   548→            if ((dex[i] & 0xff) != 0x27) continue;
   549→            // Extra confidence check: the 6 bytes before should be the
   550→            // invoke-direct for IOException.<init>(String,Throwable) which
   551→            // has 3 arguments, encoded as opcode=0x70, count|0=0x30.
   552→            boolean precededByInvokeDirect =
   553→                    i >= 6
   554→                    && (dex[i - 6] & 0xff) == 0x70   // invoke-direct opcode
   555→                    && (dex[i - 5] & 0xff) == 0x30;  // arg-count nibble = 3
   556→            if (precededByInvokeDirect) { throwOff = i; break; }
   557→        }
   558→        if (throwOff < 0) {
   559→            // Retry without the invoke-direct check (some compiler variants).
   560→            for (int i = csOff + 4; i < limit; i++) {
   561→                if ((dex[i] & 0xff) == 0x27) { throwOff = i; break; }
   562→            }
   563→        }
   564→        if (throwOff < 0) {
   565→            Log.w(TAG, "applyPatch: throw instruction not found after 'stat failed: '");
   566→            return -1;
   567→        }
   568→
   569→        // ── Step 6: apply the patch ───────────────────────────────────────────
   570→        // Replace  throw v_ioe      [0x27 0xNN]
   571→        // with     const/4 v_stat, 0 [0x12 0x0M]  (M = stat register, value = 0)
   572→        dex[throwOff]     = 0x12;
   573→        dex[throwOff + 1] = (byte) (statReg & 0x0f); // value=0 in high nibble, reg in low nibble
   574→
   575→        Log.d(TAG, "applyPatch: throw→const/4 v" + statReg
   576→                + " at 0x" + Integer.toHexString(throwOff)
   577→                + " (move-exception@0x" + Integer.toHexString(meOff)
   578→                + ", v_stat=v" + statReg + ")");
   579→        return throwOff;
   580→    }

   582→    /**
   583→     * Re-computes the Adler-32 checksum (bytes 8–11) and SHA-1 signature
   584→     * (bytes 12–31) of a DEX file in-place after patching.
   585→     *
   586→     * <p>DEX layout:
   587→     * <ul>
   588→     *   <li>bytes  0– 7  magic</li>
   589→     *   <li>bytes  8–11  Adler-32 of bytes 12..end</li>
   590→     *   <li>bytes 12–31  SHA-1   of bytes 32..end</li>
   591→     *   <li>bytes 32–35  file_size …</li>
   592→     * </ul>
   593→     */
   594→    private static void recomputeDexChecksums(byte[] dex)
   595→            throws java.security.NoSuchAlgorithmException {
   596→        // SHA-1 of bytes[32 .. end]
   597→        MessageDigest sha1 = MessageDigest.getInstance("SHA-1");
   598→        sha1.update(dex, 32, dex.length - 32);
   599→        byte[] digest = sha1.digest();
   600→        System.arraycopy(digest, 0, dex, 12, 20);
   601→
   602→        // Adler-32 of bytes[12 .. end]  (which now includes the updated SHA-1)
   603→        Adler32 adler = new Adler32();
   604→        adler.update(dex, 12, dex.length - 12);
   605→        int checksum = (int) adler.getValue();
   606→        dex[8]  = (byte)  (checksum         & 0xff);
   607→        dex[9]  = (byte) ((checksum >>  8)  & 0xff);
   608→        dex[10] = (byte) ((checksum >> 16)  & 0xff);
   609→        dex[11] = (byte) ((checksum >> 24)  & 0xff);
   610→    }

   612→    /**
   613→     * Replaces {@code classes.dex} inside {@code jar} with {@code patchedDex},
   614→     * preserving all other ZIP entries and their compression settings.
   615→     *
   616→     * @return {@code true} on success.
   617→     */
   618→    private static boolean replaceClassesDexInJar(File jar, byte[] patchedDex) {
   619→        File tmp = new File(jar.getParent(), jar.getName() + ".twoyi_tmp");
   620→        try (ZipFile  zin  = new ZipFile(jar);
   621→             ZipOutputStream zout = new ZipOutputStream(new FileOutputStream(tmp))) {
   622→
   623→            Enumeration<? extends ZipEntry> entries = zin.entries();
   624→            while (entries.hasMoreElements()) {
   625→                ZipEntry src = entries.nextElement();
   626→                byte[] data;
   627→                if ("classes.dex".equals(src.getName())) {
   628→                    data = patchedDex;
   629→                } else {
   630→                    try (InputStream in = zin.getInputStream(src)) {
   631→                        ByteArrayOutputStream baos =
   632→                                new ByteArrayOutputStream((int) Math.max(src.getSize(), 0));
   633→                        byte[] buf = new byte[8192]; int n;
   634→                        while ((n = in.read(buf)) >= 0) baos.write(buf, 0, n);
   635→                        data = baos.toByteArray();
   636→                    }
   637→                }
   638→
   639→                ZipEntry dst = new ZipEntry(src.getName());
   640→                if (src.getMethod() == ZipEntry.STORED) {
   641→                    // Must supply size/CRC for STORED entries.
   642→                    dst.setMethod(ZipEntry.STORED);
   643→                    dst.setSize(data.length);
   644→                    dst.setCompressedSize(data.length);
   645→                    CRC32 crc = new CRC32();
   646→                    crc.update(data);
   647→                    dst.setCrc(crc.getValue());
   648→                } else {
   649→                    dst.setMethod(ZipEntry.DEFLATED);
   650→                }
   651→                zout.putNextEntry(dst);
   652→                zout.write(data);
   653→                zout.closeEntry();
   654→            }
   655→            zout.finish();
   656→
   657→        } catch (IOException e) {
   658→            Log.e(TAG, "replaceClassesDexInJar: IOException", e);
   659→            tmp.delete();
   660→            return false;
   661→        }
   662→
   663→        try {
   664→            Files.move(tmp.toPath(), jar.toPath(), StandardCopyOption.REPLACE_EXISTING);
   665→            return true;
   666→        } catch (IOException e) {
   667→            Log.e(TAG, "replaceClassesDexInJar: move failed", e);
   668→            tmp.delete();
   669→            return false;
   670→        }
   671→    }

   683→    /** Recursively deletes files whose names contain {@code pattern}. */
   684→    private static void deleteMatchingFiles(File dir, String pattern) {
   685→        if (dir == null || !dir.isDirectory()) return;
   686→        File[] children = dir.listFiles();
   687→        if (children == null) return;
   688→        for (File child : children) {
   689→            if (child.isDirectory()) {
   690→                deleteMatchingFiles(child, pattern);
   691→            } else if (child.getName().contains(pattern)) {
   692→                if (!child.delete()) {
   693→                    Log.w(TAG, "deleteMatchingFiles: could not delete " + child);
   694→                }
   695→            }
   696→        }
   697→    }

   713→    /** Reads a 4-byte little-endian signed integer from {@code buf[off..off+3]}. */
   714→    private static int readInt32LE(byte[] buf, int off) {
   715→        return  (buf[off]     & 0xff)
   716→             | ((buf[off + 1] & 0xff) <<  8)
   717→             | ((buf[off + 2] & 0xff) << 16)
   718→             | ((buf[off + 3] & 0xff) << 24);
   719→    }
   720→
   721→    /**
   722→     * Reads an unsigned LEB128 (ULEB128) value from {@code buf} starting at
   723→     * {@code off} and returns the decoded integer.
   724→     */
   725→    private static int readUleb128(byte[] buf, int off) {
   726→        int result = 0, shift = 0;
   727→        while (off < buf.length) {
   728→            int b = buf[off++] & 0xff;
   729→            result |= (b & 0x7f) << shift;
   730→            if ((b & 0x80) == 0) break;
   731→            shift += 7;
   732→        }
   733→        return result;
   734→    }
   735→
   736→    /**
   737→     * Returns the number of bytes required to encode {@code value} as a
   738→     * ULEB128 integer.
   739→     */
   740→    private static int uleb128Size(int value) {
   741→        int size = 1;
   742→        while ((value & ~0x7f) != 0) { value >>>= 7; size++; }
   743→        return size;
   744→    }
