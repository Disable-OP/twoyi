    12→//! Twoyi kernel-replacement daemon — the Rust port of VM's `libkr64.so`.
    13→//!
    14→//! See `VM_KR64_ANALYSIS.md` and `GSI_BOOT_PLAN.md` for the full
    15→//! background. The short version: twoyi needs a userspace daemon that
    16→//! materialises a virtual `/dev` tree for the guest Android, installs
    17→//! a seccomp filter that traps "dangerous" syscalls, emulates `/proc`
    18→//! and `/sys`, manages the per-VM mount namespace, and exec's the
    19→//! guest's `/system/bin/init`. This crate is that daemon.
    20→//!
    21→//! # Build
    22→//!
    23→//! The crate builds as BOTH a cdylib (`libkr64.so` — directly
    24→//! executable via the PIE hack) AND a regular binary (`kr64`). See
    25→//! `Cargo.toml` and `build.rs` for details.
    26→//!
    27→//! # Usage
    28→//!
    29→//! ```sh
    30→//! # Direct execution via the PIE-hacked cdylib:
    31→//! ./libkr64.so --rootfs /data/.../fs --data-dir /data/... --vmid 0
    32→//!
    33→//! # Or via the regular binary:
    34→//! ./kr64 --rootfs /data/.../fs --data-dir /data/... --vmid 0
    35→//! ```
    36→//!
    37→//! # Module layout
    38→//!
    39→//! * [`devices`]    — virtual `/dev` device creation (qemu_pipe,
    40→//!                    touch, key, event, gb, gb2).
    41→//! * [`binder`]     — per-VM `/vm{id}/dev/binder` Unix socket + binder
    42→//!                    transaction proxy (skeleton; see
    43→//!                    `download/BINDER_SKELETON.md`).
    44→//! * [`audio`]      — virtual `/dev/audio` Unix socket + bidirectional
    45→//!                    PCM pump (playback + capture skeleton; see
    46→//!                    `download/AUDIO_SENSOR_HAL.md`).
    47→//! * [`sensors`]    — virtual `/dev/sensors` Unix socket + multiplexed
    48→//!                    12-sensor HAL (accel/mag/gyro/... skeleton; see
    49→//!                    `download/AUDIO_SENSOR_HAL.md`).
    50→//! * [`battery`]    — virtual `/sys/class/power_supply/battery` file tree
    51→//!                    + 30 s refresh thread (file-based, no socket; see
    52→//!                    `download/BATTERY_IMPL.md`).
    53→//! * [`seccomp`]    — BPF seccomp filter + SIGSYS handler.
    54→//! * [`proc_emu`]   — synthesised `/proc` tree (version, cpuinfo,
    55→//!                    meminfo, cmdline, self/, sys/).
    56→//! * [`mount_mgr`]  — `unshare(CLONE_NEWNS)` + `pivot_root` + tmpfs
    57→//!                    mounts for /dev, /proc, /sys, /system, /vendor.

   295→/// Run the daemon. Returns the exit code.
   296→///
   297→/// This is the shared entry point called by both:
   298→///   * `kr64_main` (the cdylib's PIE entry point, see below), and
   299→///   * `main()` in `src/main.rs` (the bin target).
   300→///
   301→/// Steps:
   302→///   1. Parse args.
   303→///   2. Create all virtual `/dev` devices (qemu_pipe, touch, key,
   304→///      event, gb, gb2).
   305→///   3. Populate `/proc` (version, cpuinfo, meminfo, cmdline, …).
   306→///   4. Set up mount namespace (unshare + bind mounts + tmpfs mounts).
   307→///   5. Fork:
   308→///      - Child: pivot_root → install seccomp → exec /system/bin/init.
   309→///      - Parent: run the device-accept loop (spawns one thread per
   310→///        device socket; for the MVP each thread just accepts and
   311→///        echoes).
   312→///   6. Wait for the child (the guest init) to exit; propagate its
   313→///      exit code.
   314→pub fn run<I: IntoIterator<Item = String>>(args: I) -> i32 {
   315→    let cfg = match parse_args(args) {
   316→        Ok(c) => c,
   317→        Err(e) => {
   318→            // If --help, print to stdout and exit 0; otherwise stderr
   319→            // and exit 2.
   320→            if e.starts_with("twoyi kr64") {
   321→                println!("{}", e);
   322→                return 0;
   323→            }
   324→            eprintln!("kr64: {}", e);
   325→            return 2;
   326→        }
   327→    };
   328→
   329→    info!("[KR64] starting daemon with config: {:?}", cfg);
   330→
   331→    // ---------------------------------------------------------------
   332→    // Step 2: create virtual /dev devices.
   333→    // ---------------------------------------------------------------
   334→    let device_set = match devices::create_all_devices(&cfg.rootfs, &cfg.data_dir) {
   335→        Ok(d) => d,
   336→        Err(e) => {
   337→            error!("[KR64] failed to create devices: {}", e);
   338→            return 1;
   339→        }
   340→    };
   341→
   342→    // Create the marker files the guest init looks for.
   343→    if let Err(e) = devices::create_coldboot_done_marker(&cfg.rootfs) {
   344→        warning!("[KR64] failed to create .coldboot_done marker: {}", e);
   345→    }
   346→    if let Err(e) = devices::create_busybox_marker(&cfg.rootfs) {
   347→        warning!("[KR64] failed to create .busybox marker: {}", e);
   348→    }
   349→
   350→    // ---------------------------------------------------------------
   351→    // Step 2.5: create the per-VM binder device + spawn the binder
   352→    // proxy. See `download/BINDER_SKELETON.md` and
   353→    // `app/rs/kr64/src/binder.rs` for the full story. The short
   354→    // version: this creates `{rootfs}/vm{id}/dev/binder` as a Unix
   355→    // socket + a `{rootfs}/dev/binder` symlink, and spawns an
   356→    // accept-thread + worker-pool that dispatches BINDER_* ioctls
   357→    // from the guest. The handle is held for the lifetime of `run()`
   358→    // so the proxy is shut down when the guest exits.
   359→    // ---------------------------------------------------------------
   360→    let _binder_handle = match binder::create_binder_device(&cfg.rootfs, cfg.vmid)
   361→        .and_then(|path| binder::BinderProxy::new(cfg.vmid, &path))
   362→        .and_then(|proxy| proxy.spawn())
   363→    {
   364→        Ok(h) => {
   365→            info!("[KR64] binder proxy listening at {} (vm{})", h.path(), cfg.vmid);
   366→            Some(h)
   367→        }
   368→        Err(e) => {
   369→            // Non-fatal: the guest can still boot against the host's
   370→            // /dev/binder if it's bind-mounted in (the current twoyi
   371→            // approach). The binder proxy is only needed for full
   372→            // service-manager virtualisation.
   373→            warning!("[KR64] failed to start binder proxy: {} — falling back to host binder", e);
   374→            None
   375→        }
   376→    };
   377→
   378→    // ---------------------------------------------------------------
   379→    // Step 2.6: create the audio device + spawn the accept/pump
   380→    // thread. See `download/AUDIO_SENSOR_HAL.md` and
   381→    // `app/rs/kr64/src/audio.rs` for the full story. The short
   382→    // version: this creates `{rootfs}/dev/audio` as a Unix socket
   383→    // and spawns an accept-thread + worker-pool that reads a
   384→    // 16-byte header per connection and pumps raw PCM in both
   385→    // directions. The actual AudioTrack/AudioRecord integration is
   386→    // stubbed (no JNI yet) — the pump compiles and exercises the
   387→    // protocol but produces no sound until the Java side is wired
   388→    // up in a follow-up task.
   389→    // ---------------------------------------------------------------
   390→    let _audio_handle = match audio::create_audio_device(&cfg.rootfs)
   391→        .and_then(|dev| dev.spawn())
   392→    {
   393→        Ok(h) => {
   394→            info!("[KR64] audio device listening at {}", h.path());
   395→            Some(h)
   396→        }
   397→        Err(e) => {
   398→            // Non-fatal: the guest can still boot without sound —
   399→            // AudioFlinger's connect() to /dev/audio will fail and
   400→            // the guest's audio HAL will fall back to silence / a
   401→            // null output. Sound is the user's primary use case
   402→            // though, so this warning is worth surfacing.
   403→            warning!("[KR64] failed to start audio device: {} — guest will have no sound", e);
   404→            None
   405→        }
   406→    };
   407→
   408→    // ---------------------------------------------------------------
   409→    // Step 2.7: create the sensor device + spawn the accept/pump
   410→    // thread. See `download/AUDIO_SENSOR_HAL.md` and
   411→    // `app/rs/kr64/src/sensors.rs` for the full story. The short
   412→    // version: this creates `{rootfs}/dev/sensors` as a Unix socket
   413→    // and spawns an accept-thread + worker-pool that reads 12-byte
   414→    // control requests (ENABLE/DISABLE/CHECK_SUPPORT/SET_DELAY) and
   415→    // pushes 24-byte SensorEvent records to the guest when sensors
   416→    // are enabled. The actual SensorManager integration is stubbed
   417→    // (no JNI yet) — the control loop replies false to every
   418→    // CHECK_SUPPORT and the pump never produces events. This is the
   419→    // deliberate "skeleton" boundary, mirroring audio.rs.
   420→    // ---------------------------------------------------------------
   421→    let _sensor_handle = match sensors::create_sensor_device(&cfg.rootfs)
   422→        .and_then(|dev| dev.spawn())
   423→    {
   424→        Ok(h) => {
   425→            info!("[KR64] sensor device listening at {}", h.path());
   426→            Some(h)
   427→        }
   428→        Err(e) => {
   429→            // Non-fatal: the guest can still boot without sensors —
   430→            // the guest's sensor HAL will see "no sensors available"
   431→            // and `SensorManager.getDefaultSensor()` will return null.
   432→            // Apps that hard-require a sensor (e.g. compass apps)
   433→            // will crash, but the boot proceeds.
   434→            warning!("[KR64] failed to start sensor device: {} — guest will have no sensors", e);
   435→            None
   436→        }
   437→    };
   438→
   439→    // ---------------------------------------------------------------
   440→    // Step 2.8: materialise the virtual battery sysfs tree + spawn
   441→    // the 30 s refresh thread. See `download/BATTERY_IMPL.md` and
   442→    // `app/rs/kr64/src/battery.rs` for the full story. The short
   443→    // version: this creates `{rootfs}/sys/class/power_supply/battery/`
   444→    // with seven files (capacity, status, charging, voltage,
   445→    // temperature, technology, health) populated from the (stubbed)
   446→    // JNI up-calls, then spawns a `kr64-battery-refresh` thread that
   447→    // re-writes them every 30 s. Failure is non-fatal — the guest
   448→    // can still boot without a battery sysfs (its `BatteryService`
   449→    // will just report "unknown" / fall back to defaults), but every
   450→    // real device has a battery so we warn loudly.
   451→    // ---------------------------------------------------------------
   452→    let _battery_handle = match battery::BatteryDevice::new(&cfg.rootfs)
   453→        .and_then(|dev| dev.spawn())
   454→    {
   455→        Ok(h) => {
   456→            info!("[KR64] battery sysfs materialised at {}/sys/class/power_supply/battery",
   457→                  cfg.rootfs);
   458→            Some(h)
   459→        }
   460→        Err(e) => {
   461→            warning!("[KR64] failed to start battery HAL: {} — guest will see no battery", e);
   462→            None
   463→        }
   464→    };
   465→
   466→    // ---------------------------------------------------------------
   467→    // Step 3: populate /proc.
   468→    // ---------------------------------------------------------------
   469→    // We synthesise for an 8-core, 4 GB guest by default. A production
   470→    // version would query the host's CPU count and memory size and
   471→    // scale these accordingly.
   472→    let cpu_count = 8u32;
   473→    let mem_mb = 4096u64;
   474→    if let Err(e) = proc_emu::populate_proc(&cfg.rootfs, cpu_count, mem_mb) {
   475→        warning!("[KR64] failed to populate /proc: {}", e);
   476→    }
   477→
   478→    // ---------------------------------------------------------------
   479→    // Step 4: set up mount namespace + bind mounts + tmpfs.
   480→    // ---------------------------------------------------------------
   481→    // NOTE: We do NOT do this yet — the mount setup has to happen
   482→    // inside the forked child, because once we pivot_root we lose
   483→    // access to the host's paths (and the parent needs to keep
   484→    // accepting connections on the device sockets it bound above).
   485→    //
   486→    // The flow is:
   487→    //   parent: bind device sockets (already done above)
   488→    //   parent: fork()
   489→    //     child: mount_mgr::setup_mounts(cfg)
   490→    //     child: seccomp::install()
   491→    //     child: execve(/system/bin/init)
   492→    //   parent: accept loop on device sockets
   493→
   494→    // ---------------------------------------------------------------
   495→    // Step 5: fork + exec the guest.
   496→    // ---------------------------------------------------------------
   497→    info!("[KR64] forking guest process");
   498→    let pid = unsafe { libc::fork() };
   499→    if pid < 0 {
   500→        let e = std::io::Error::last_os_error();
   501→        error!("[KR64] fork failed: {}", e);
   502→        return 1;
   503→    }
   504→
   505→    if pid == 0 {
   506→        // ----- CHILD (will become the guest init) -----
   507→        info!("[KR64][child] setting up mounts");
   508→        let mount_cfg = mount_mgr::MountConfig {
   509→            rootfs:           cfg.rootfs.clone(),
   510→            rom_dir:          cfg.rom_dir.clone(),
   511→            use_namespaces:   cfg.use_namespaces,
   512→            read_only_rom:    cfg.read_only_rom,
   513→        };
   514→        if let Err(e) = mount_mgr::setup_mounts(&mount_cfg) {
   515→            error!("[KR64][child] mount setup failed: {}", e);
   516→            unsafe { libc::_exit(1) };
   517→        }
   518→
   519→        // Install seccomp filter on the guest process. This MUST be
   520→        // done after mount setup (otherwise the mount syscalls would
   521→        // be trapped and we couldn't set up the rootfs).
   522→        if cfg.install_seccomp {
   523→            info!("[KR64][child] installing seccomp filter");
   524→            if let Err(e) = seccomp::install() {
   525→                error!("[KR64][child] seccomp install failed: {}", e);
   526→                // Continue anyway — the guest will see the host's
   527→                // unfiltered syscalls, which is insecure but lets
   528→                // debugging proceed.
   529→            }
   530→        }
   531→
   532→        // Exec the guest init.
   533→        info!("[KR64][child] exec'ing guest init at {}", cfg.init_path);
   534→        let init_cstr = CString::new(cfg.init_path.as_str()).unwrap();
   535→        let argv0 = CString::new(cfg.init_path.as_str()).unwrap();
   536→        let argv: [*const libc::c_char; 2] = [argv0.as_ptr(), std::ptr::null()];
   537→        let envp: [*const libc::c_char; 1] = [std::ptr::null()];
   538→
   539→        // Set a few environment variables the guest init may look for.
   540→        // (execve replaces envp entirely, so we have to construct it.)
   541→        // For the MVP we just pass an empty envp — the guest's init
   542→        // builds its own environment from /proc/cmdline + properties.
   543→        let r = unsafe { libc::execve(init_cstr.as_ptr(), argv.as_ptr(), envp.as_ptr()) };
   544→        let e = std::io::Error::last_os_error();
   545→        error!("[KR64][child] execve({}) failed (r={}): {}", cfg.init_path, r, e);
   546→        unsafe { libc::_exit(127) };
   547→    }
   548→
   549→    // ----- PARENT (the daemon) -----
   550→    info!("[KR64][parent] guest pid = {}", pid);
   551→
   552→    // Spawn one accept thread per device socket. For the MVP each
   553→    // thread just accepts connections and immediately closes them
   554→    // (echoing a single byte so the guest sees SOME response). The
   555→    // production version will dispatch to per-device handlers:
   556→    //   qemu_pipe → openglrenderer::pipe
   557→    //   touch     → input::touch_server
   558→    //   key       → input::key_server
   559→    //   event     → TwoyiSocketServer (event IPC)
   560→    //   gb/gb2    → openglrenderer::gralloc
   561→    spawn_accept_thread(device_set.qemu_pipe, "qemu_pipe");
   562→    spawn_accept_thread(device_set.touch,     "touch");
   563→    spawn_accept_thread(device_set.key,       "key");
   564→    spawn_accept_thread(device_set.event,     "event");
   565→    spawn_accept_thread(device_set.gb.gb,     "gb");
   566→    spawn_accept_thread(device_set.gb.gb2,    "gb2");
   567→
   568→    // ---------------------------------------------------------------
   569→    // Step 6: wait for the guest to exit.
   570→    // ---------------------------------------------------------------
   571→    let mut status: libc::c_int = 0;
   572→    loop {
   573→        let waited = unsafe { libc::waitpid(pid, &mut status, 0) };
   574→        if waited == -1 {
   575→            let e = std::io::Error::last_os_error();
   576→            if e.raw_os_error() == Some(libc::EINTR) {
   577→                continue;
   578→            }
   579→            error!("[KR64][parent] waitpid failed: {}", e);
   580→            return 1;
   581→        }
   582→        break;
   583→    }
   584→
   585→    if libc::WIFEXITED(status) {
   586→        let code = libc::WEXITSTATUS(status);
   587→        info!("[KR64][parent] guest exited with status {}", code);
   588→        return code;
   589→    }
   590→    if libc::WIFSIGNALED(status) {
   591→        let sig = libc::WTERMSIG(status);
   592→        warning!("[KR64][parent] guest killed by signal {}", sig);
   593→        return 128 + sig;
   594→    }
   595→
   596→    warning!("[KR64][parent] guest waitpid returned unexpected status {}", status);
   597→    1
   598→}
   599→
   600→/// Spawn a thread that accepts connections on the given device socket
   601→/// and immediately closes them (MVP placeholder).
   602→///
   603→/// The thread takes ownership of the underlying `UnixListener` (so
   604→/// the parent `DeviceSocket` is consumed). For the production version
   605→/// this will be replaced with per-device handler dispatch.
   606→fn spawn_accept_thread(mut dev: devices::DeviceSocket, name: &'static str) {
   607→    let listener = match dev.take_listener() {
   608→        Some(l) => l,
   609→        None => {
   610→            warning!("[KR64] cannot spawn accept thread for {}: listener already taken", name);
   611→            return;
   612→        }
   613→    };
   614→    // Make the listening socket non-blocking so the thread can poll
   615→    // for shutdown signals (in the MVP we just loop forever).
   616→    let fd = listener.as_raw_fd();
   617→    let _ = unsafe { libc::fcntl(fd, libc::F_SETFL, libc::O_NONBLOCK) };
   618→
   619→    std::thread::Builder::new()
   620→        .name(format!("kr64-accept-{}", name))
   621→        .spawn(move || {
   622→            info!("[KR64][{}] accept thread started (fd={})", name, fd);
   623→            loop {
   624→                match listener.accept() {
   625→                    Ok((mut stream, _addr)) => {
   626→                        info!("[KR64][{}] client connected", name);
   627→                        // Echo a single byte so the guest sees SOME response.
   628→                        // (Many of the device protocols expect a handshake
   629→                        // byte — e.g. the touch device sends a device_info
   630→                        // struct on connect, which the guest reads before
   631→                        // sending anything. The production version will
   632→                        // dispatch to the right handler.)
   633→                        use std::io::Write;
   634→                        let _ = stream.write_all(&[0u8]);
   635→                    }
   636→                    Err(e) if e.kind() == std::io::ErrorKind::WouldBlock => {
   637→                        // No pending connection — sleep briefly to avoid
   638→                        // spinning. (Real implementation would use epoll.)
   639→                        std::thread::sleep(std::time::Duration::from_millis(50));
   640→                    }
   641→                    Err(e) => {
   642→                        warning!("[KR64][{}] accept error: {}", name, e);
   643→                        std::thread::sleep(std::time::Duration::from_millis(100));
   644→                    }
   645→                }
   646→            }
   647→        })
   648→        .expect("spawn kr64 accept thread");
   649→}

   651→// ============================================================================
   652→// cdylib entry point — used when `libkr64.so` is exec'd directly via
   653→// the PIE hack. The ELF entry point is set to `kr64_main` via the
   654→// `-Wl,-e,kr64_main` link flag in build.rs / .cargo/config.toml.
   655→// ============================================================================
   656→
   657→/// Entry point for the cdylib (libkr64.so). Mirrors the C `main`
   658→/// signature `(argc, argv) → int` so the standard C runtime can call
   659→/// it.
   660→///
   661→/// When the kernel exec's `libkr64.so`, the dynamic linker loads it,
   662→/// runs `.init_array`, and jumps to `kr64_main` (because we set
   663→/// `-Wl,-e,kr64_main`). The args come from the kernel's stack as
   664→/// `argc` + `argv[]`, just like a regular C `main`.
   665→#[no_mangle]
   666→pub extern "C" fn kr64_main(argc: libc::c_int, argv: *const *const libc::c_char) -> libc::c_int {
   667→    // Convert C argv → Rust Vec<String>.
   668→    let args: Vec<String> = if argc <= 0 || argv.is_null() {
   669→        Vec::new()
   670→    } else {
   671→        (0..argc as isize)
   672→            .map(|i| unsafe {
   673→                let p = *argv.offset(i);
   674→                if p.is_null() { String::new() }
   675→                else { std::ffi::CStr::from_ptr(p).to_string_lossy().to_string() }
   676→            })
   677→            .collect()
   678→    };
   679→
   680→    run(args)
   681→}
   682→
   683→// Reference the interp symbol from interp.c (built by build.rs) so the
   684→// linker keeps the `.interp` section in the final libkr64.so. Without
   685→// this reference, the linker may GC the `.interp` section as unused,
   686→// and the resulting .so won't be directly executable.
   687→extern "C" {
   688→    #[link_name = "interp"]
   689→    static INTERP: [u8; 0];
   690→}
   691→
   692→#[used]
   693→static INTERP_REF: &'static [u8; 0] = unsafe { &INTERP };
