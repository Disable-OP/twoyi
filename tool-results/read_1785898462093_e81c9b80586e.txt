    12→//! Twoyi kernel-replacement daemon — the Rust port of VM's `libkr64.so`.
    13→//!
    14→//! See `VM_KR64_ANALYSIS.md` and `GSI_BOOT_PLAN.md` for the full
    15→//! background. The short version: twoyi needs a userspace daemon that
    16→//! materialises a virtual `/dev` tree for the guest Android, installs
    17→//! a seccomp filter that traps "dangerous" syscalls, emulates `/proc`
    18→//! and `/sys`, manages the per-VM mount namespace, and exec's the
    19→//! guest's `/system/bin/init`. This crate is that daemon.
    20→//!
    21→//! # Build
    22→//!
    23→//! The crate builds as BOTH a cdylib (`libkr64.so` — directly
    24→//! executable via the PIE hack) AND a regular binary (`kr64`). See
    25→//! `Cargo.toml` and `build.rs` for details.
    26→//!
    27→//! # Usage
    28→//!
    29→//! ```sh
    30→//! # Direct execution via the PIE-hacked cdylib:
    31→//! ./libkr64.so --rootfs /data/.../fs --data-dir /data/... --vmid 0
    32→//!
    33→//! # Or via the regular binary:
    34→//! ./kr64 --rootfs /data/.../fs --data-dir /data/... --vmid 0
    35→//! ```
    36→//!

   279→/// Run the daemon. Returns the exit code.
   280→///
   281→/// This is the shared entry point called by both:
   282→///   * `kr64_main` (the cdylib's PIE entry point, see below), and
   283→///   * `main()` in `src/main.rs` (the bin target).
   284→///
   285→/// Steps:
   286→///   1. Parse args.
   287→///   2. Create all virtual `/dev` devices (qemu_pipe, touch, key,
   288→///      event, gb, gb2).
   289→///   3. Populate `/proc` (version, cpuinfo, meminfo, cmdline, …).
   290→///   4. Set up mount namespace (unshare + bind mounts + tmpfs mounts).
   291→///   5. Fork:
   292→///      - Child: pivot_root → install seccomp → exec /system/bin/init.
   293→///      - Parent: run the device-accept loop (spawns one thread per
   294→///        device socket; for the MVP each thread just accepts and
   295→///        echoes).
   296→///   6. Wait for the child (the guest init) to exit; propagate its
   297→///      exit code.
   298→pub fn run<I: IntoIterator<Item = String>>(args: I) -> i32 {
   299→    let cfg = match parse_args(args) {
   300→        Ok(c) => c,
   301→        Err(e) => {
   302→            // If --help, print to stdout and exit 0; otherwise stderr
   303→            // and exit 2.
   304→            if e.starts_with("twoyi kr64") {
   305→                println!("{}", e);
   306→                return 0;
   307→            }
   308→            eprintln!("kr64: {}", e);
   309→            return 2;
   310→        }
   311→    };
   312→
   313→    info!("[KR64] starting daemon with config: {:?}", cfg);
   314→
   315→    // ---------------------------------------------------------------
   316→    // Step 2: create virtual /dev devices.
   317→    // ---------------------------------------------------------------
   318→    let device_set = match devices::create_all_devices(&cfg.rootfs, &cfg.data_dir) {
   319→        Ok(d) => d,
   320→        Err(e) => {
   321→            error!("[KR64] failed to create devices: {}", e);
   322→            return 1;
   323→        }
   324→    };
   325→
   326→    // Create the marker files the guest init looks for.
   327→    if let Err(e) = devices::create_coldboot_done_marker(&cfg.rootfs) {
   328→        warning!("[KR64] failed to create .coldboot_done marker: {}", e);
   329→    }
   330→    if let Err(e) = devices::create_busybox_marker(&cfg.rootfs) {
   331→        warning!("[KR64] failed to create .busybox marker: {}", e);
   332→    }
   333→
   334→    // ---------------------------------------------------------------
   335→    // Step 3: populate /proc.
   336→    // ---------------------------------------------------------------
   337→    // We synthesise for an 8-core, 4 GB guest by default. A production
   338→    // version would query the host's CPU count and memory size and
   339→    // scale these accordingly.
   340→    let cpu_count = 8u32;
   341→    let mem_mb = 4096u64;
   342→    if let Err(e) = proc_emu::populate_proc(&cfg.rootfs, cpu_count, mem_mb) {
   343→        warning!("[KR64] failed to populate /proc: {}", e);
   344→    }
   345→
   346→    // ---------------------------------------------------------------
   347→    // Step 4: set up mount namespace + bind mounts + tmpfs.
   348→    // ---------------------------------------------------------------
   349→    // NOTE: We do NOT do this yet — the mount setup has to happen
   350→    // inside the forked child, because once we pivot_root we lose
   351→    // access to the host's paths (and the parent needs to keep
   352→    // accepting connections on the device sockets it bound above).
   353→    //
   354→    // The flow is:
   355→    //   parent: bind device sockets (already done above)
   356→    //   parent: fork()
   357→    //     child: mount_mgr::setup_mounts(cfg)
   358→    //     child: seccomp::install()
   359→    //     child: execve(/system/bin/init)
   360→    //   parent: accept loop on device sockets
   361→
   362→    // ---------------------------------------------------------------
   363→    // Step 5: fork + exec the guest.
   364→    // ---------------------------------------------------------------
   365→    info!("[KR64] forking guest process");
   366→    let pid = unsafe { libc::fork() };
   367→    if pid < 0 {
   368→        let e = std::io::Error::last_os_error();
   369→        error!("[KR64] fork failed: {}", e);
   370→        return 1;
   371→    }
   372→
   373→    if pid == 0 {
   374→        // ----- CHILD (will become the guest init) -----
   375→        info!("[KR64][child] setting up mounts");
   376→        let mount_cfg = mount_mgr::MountConfig {
   377→            rootfs:           cfg.rootfs.clone(),
   378→            rom_dir:          cfg.rom_dir.clone(),
   379→            use_namespaces:   cfg.use_namespaces,
   380→            read_only_rom:    cfg.read_only_rom,
   381→        };
   382→        if let Err(e) = mount_mgr::setup_mounts(&mount_cfg) {
   383→            error!("[KR64][child] mount setup failed: {}", e);
   384→            unsafe { libc::_exit(1) };
   385→        }
   386→
   387→        // Install seccomp filter on the guest process. This MUST be
   388→        // done after mount setup (otherwise the mount syscalls would
   389→        // be trapped and we couldn't set up the rootfs).
   390→        if cfg.install_seccomp {
   391→            info!("[KR64][child] installing seccomp filter");
   392→            if let Err(e) = seccomp::install() {
   393→                error!("[KR64][child] seccomp install failed: {}", e);
   394→                // Continue anyway — the guest will see the host's
   395→                // unfiltered syscalls, which is insecure but lets
   396→                // debugging proceed.
   397→            }
   398→        }
   399→
   400→        // Exec the guest init.
   401→        info!("[KR64][child] exec'ing guest init at {}", cfg.init_path);
   402→        let init_cstr = CString::new(cfg.init_path.as_str()).unwrap();
   403→        let argv0 = CString::new(cfg.init_path.as_str()).unwrap();
   404→        let argv: [*const libc::c_char; 2] = [argv0.as_ptr(), std::ptr::null()];
   405→        let envp: [*const libc::c_char; 1] = [std::ptr::null()];
   406→
   407→        // Set a few environment variables the guest init may look for.
   408→        // (execve replaces envp entirely, so we have to construct it.)
   409→        // For the MVP we just pass an empty envp — the guest's init
   410→        // builds its own environment from /proc/cmdline + properties.
   411→        let r = unsafe { libc::execve(init_cstr.as_ptr(), argv.as_ptr(), envp.as_ptr()) };
   412→        let e = std::io::Error::last_os_error();
   413→        error!("[KR64][child] execve({}) failed (r={}): {}", cfg.init_path, r, e);
   414→        unsafe { libc::_exit(127) };
   415→    }
   416→
   417→    // ----- PARENT (the daemon) -----
   418→    info!("[KR64][parent] guest pid = {}", pid);
   419→
   420→    // Spawn one accept thread per device socket. For the MVP each
   421→    // thread just accepts connections and immediately closes them
   422→    // (echoing a single byte so the guest sees SOME response). The
   423→    // production version will dispatch to per-device handlers:
   424→    //   qemu_pipe → openglrenderer::pipe
   425→    //   touch     → input::touch_server
   426→    //   key       → input::key_server
   427→    //   event     → TwoyiSocketServer (event IPC)
   428→    //   gb/gb2    → openglrenderer::gralloc
   429→    spawn_accept_thread(device_set.qemu_pipe, "qemu_pipe");
   430→    spawn_accept_thread(device_set.touch,     "touch");
   431→    spawn_accept_thread(device_set.key,       "key");
   432→    spawn_accept_thread(device_set.event,     "event");
   433→    spawn_accept_thread(device_set.gb.gb,     "gb");
   434→    spawn_accept_thread(device_set.gb.gb2,    "gb2");
   435→
   436→    // ---------------------------------------------------------------
   437→    // Step 6: wait for the guest to exit.
   438→    // ---------------------------------------------------------------
   439→    let mut status: libc::c_int = 0;
   440→    loop {
   441→        let waited = unsafe { libc::waitpid(pid, &mut status, 0) };
   442→        if waited == -1 {
   443→            let e = std::io::Error::last_os_error();
   444→            if e.raw_os_error() == Some(libc::EINTR) {
   445→                continue;
   446→            }
   447→            error!("[KR64][parent] waitpid failed: {}", e);
   448→            return 1;
   449→        }
   450→        break;
   451→    }
   452→
   453→    if libc::WIFEXITED(status) {
   454→        let code = libc::WEXITSTATUS(status);
   455→        info!("[KR64][parent] guest exited with status {}", code);
   456→        return code;
   457→    }
   458→    if libc::WIFSIGNALED(status) {
   459→        let sig = libc::WTERMSIG(status);
   460→        warning!("[KR64][parent] guest killed by signal {}", sig);
   461→        return 128 + sig;
   462→    }
   463→
   464→    warning!("[KR64][parent] guest waitpid returned unexpected status {}", status);
   465→    1
   466→}

   468→/// Spawn a thread that accepts connections on the given device socket
   469→/// and immediately closes them (MVP placeholder).
   470→///
   471→/// The thread takes ownership of the underlying `UnixListener` (so
   472→/// the parent `DeviceSocket` is consumed). For the production version
   473→/// this will be replaced with per-device handler dispatch.
   474→fn spawn_accept_thread(mut dev: devices::DeviceSocket, name: &'static str) {
   475→    let listener = match dev.take_listener() {
   476→        Some(l) => l,
   477→        None => {
   478→            warning!("[KR64] cannot spawn accept thread for {}: listener already taken", name);
   479→            return;
   480→        }
   481→    };
   482→    // Make the listening socket non-blocking so the thread can poll
   483→    // for shutdown signals (in the MVP we just loop forever).
   484→    let fd = listener.as_raw_fd();
   485→    let _ = unsafe { libc::fcntl(fd, libc::F_SETFL, libc::O_NONBLOCK) };
   486→
   487→    std::thread::Builder::new()
   488→        .name(format!("kr64-accept-{}", name))
   489→        .spawn(move || {
   490→            info!("[KR64][{}] accept thread started (fd={})", name, fd);
   491→            loop {
   492→                match listener.accept() {
   493→                    Ok((mut stream, _addr)) => {
   494→                        info!("[KR64][{}] client connected", name);
   495→                        // Echo a single byte so the guest sees SOME response.
   496→                        // (Many of the device protocols expect a handshake
   497→                        // byte — e.g. the touch device sends a device_info
   498→                        // struct on connect, which the guest reads before
   499→                        // sending anything. The production version will
   500→                        // dispatch to the right handler.)
   501→                        use std::io::Write;
   502→                        let _ = stream.write_all(&[0u8]);
   503→                    }
   504→                    Err(e) if e.kind() == std::io::ErrorKind::WouldBlock => {
   505→                        // No pending connection — sleep briefly to avoid
   506→                        // spinning. (Real implementation would use epoll.)
   507→                        std::thread::sleep(std::time::Duration::from_millis(50));
   508→                    }
   509→                    Err(e) => {
   510→                        warning!("[KR64][{}] accept error: {}", name, e);
   511→                        std::thread::sleep(std::time::Duration::from_millis(100));
   512→                    }
   513→                }
   514→            }
   515→        })
   516→        .expect("spawn kr64 accept thread");
   517→}

   519→// ============================================================================
   520→// cdylib entry point — used when `libkr64.so` is exec'd directly via
   521→// the PIE hack. The ELF entry point is set to `kr64_main` via the
   522→// `-Wl,-e,kr64_main` link flag in build.rs / .cargo/config.toml.
   523→// ============================================================================
   524→
   525→/// Entry point for the cdylib (libkr64.so). Mirrors the C `main`
   526→/// signature `(argc, argv) → int` so the standard C runtime can call
   527→/// it.
   528→///
   529→/// When the kernel exec's `libkr64.so`, the dynamic linker loads it,
   530→/// runs `.init_array`, and jumps to `kr64_main` (because we set
   531→/// `-Wl,-e,kr64_main`). The args come from the kernel's stack as
   532→/// `argc` + `argv[]`, just like a regular C `main`.
   533→#[no_mangle]
   534→pub extern "C" fn kr64_main(argc: libc::c_int, argv: *const *const libc::c_char) -> libc::c_int {
   535→    // Convert C argv → Rust Vec<String>.
   536→    let args: Vec<String> = if argc <= 0 || argv.is_null() {
   537→        Vec::new()
   538→    } else {
   539→        (0..argc as isize)
   540→            .map(|i| unsafe {
   541→                let p = *argv.offset(i);
   542→                if p.is_null() { String::new() }
   543→                else { std::ffi::CStr::from_ptr(p).to_string_lossy().to_string() }
   544→            })
   545→            .collect()
   546→    };
   547→
   548→    run(args)
   549→}
   550→
   551→// Reference the interp symbol from interp.c (built by build.rs) so the
   552→// linker keeps the `.interp` section in the final libkr64.so. Without
   553→// this reference, the linker may GC the `.interp` section as unused,
   554→// and the resulting .so won't be directly executable.
   555→extern "C" {
   556→    #[link_name = "interp"]
   557→    static INTERP: [u8; 0];
   558→}
   559→
   560→#[used]
   561→static INTERP_REF: &'static [u8; 0] = unsafe { &INTERP };
