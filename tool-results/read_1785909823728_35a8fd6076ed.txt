    12→//! Seccomp filter + SIGSYS handler — the "kernel replacement" syscall
    13→//! interception layer.
    14→//!
    15→//! This mirrors what VM's `libkr64.so` does (see `VM_KR64_ANALYSIS.md`
    16→//! §12 and §11):
    17→//!
    18→//! 1. `prctl(PR_SET_NO_NEW_PRIVS, 1)` — required before installing a
    19→//!    seccomp filter without `CAP_SYS_ADMIN`.
    20→//! 2. `sigaction(SIGSYS, …)` — install a handler that catches trapped
    21→//!    syscalls. The string `__NR_rt_sigaction SIGSYS` was decoded from
    22→//!    `libkr64.11.so`'s `.data` (key 0x1a).
    23→//! 3. `seccomp(SECCOMP_SET_MODE_FILTER, …)` — install a BPF program
    24→//!    that traps forbidden syscalls. The BPF program returns
    25→//!    `SECCOMP_RET_TRAP` for "intercepted" syscalls (mount, umount,
    26→//!    reboot, kexec_load, init_module, etc.) and `SECCOMP_RET_ALLOW`
    27→//!    for everything else.
    28→//! 4. The SIGSYS handler logs `BLOCKED.SYSCALL.FAILED: <nr>` (string
    29→//!    decoded from libkr64.so with key 0xc9 / 0xc2) and either:
    30→//!    - Emulates the syscall (sets the return value to 0 / fake-success
    31→//!      and advances the PC past the syscall instruction), OR
    32→//!    - Kills the guest (via `_exit`).
    33→//!
    34→//! # Why `SECCOMP_RET_TRAP`, not `SECCOMP_RET_TRACE`
    35→//!
    36→//! The task spec mentions `SECCOMP_RET_TRACE`, but `SECCOMP_RET_TRACE`
    37→//! requires a ptrace tracer to be attached to the process — without
    38→//! one, the kernel falls back to `SECCOMP_RET_KILL`. The mechanism
    39→//! that actually delivers `SIGSYS` (and the one VM uses, per the
    40→//! analysis) is `SECCOMP_RET_TRAP`. We expose `SECCOMP_RET_TRACE` as
    41→//! a constant for callers that want to switch to ptrace-mode tracing
    42→//! later (e.g. for debugging the guest), but the default filter uses
    43→//! `SECCOMP_RET_TRAP`.
    44→//!
    45→//! # Why we don't trap EVERY syscall
    46→//!
    47→//! VM's filter traps a small set of dangerous syscalls (mount, reboot,
    48→//! kexec, init_module, ptrace, etc.) and lets everything else go
    49→//! through. This is the right tradeoff — trapping every syscall (and
    50→//! emulating each one in the handler) would be 100× slower and would
    51→//! require a complete kernel emulation layer. We only trap syscalls
    52→//! that:
    53→//!   (a) would let the guest escape its sandbox, OR
    54→//!   (b) need to be redirected to a userspace implementation
    55→//!       (mount → bind mount, umount → unbind, swapon → no-op, etc.)

   135→/// Build a `BPF_LD | BPF_W | BPF_ABS` instruction — load a 32-bit word
   136→/// from offset `k` of `struct seccomp_data` into the accumulator.
   137→const fn bpf_ld_abs(k: u32) -> SockFilter {
   138→    SockFilter { code: BPF_LD | BPF_W | BPF_ABS, jt: 0, jf: 0, k }

   144→    SockFilter { code: BPF_JMP | BPF_JEQ | BPF_K, jt, jf, k }

   381→/// Compile the allow/trap/kill classification into a BPF program.
   382→///
   383→/// The program structure is:
   384→/// ```text
   385→///   1. ld arch                         // load audit arch
   386→///   2. jeq EXPECTED_ARCH, jt=0, jf=N   // if wrong arch, jump to kill
   387→///   3. ld nr                           // load syscall number
   388→///   4. for each allowed syscall:
   389→///        jeq nr, jt=0, jf=1            // if match: fall to ret ALLOW
   390→///        ret ALLOW                      // else: skip to next jeq
   391→///   5. for each trapped syscall:
   392→///        jeq nr, jt=0, jf=1
   393→///        ret TRAP
   394→///   6. for each killed syscall:
   395→///        jeq nr, jt=0, jf=1
   396→///        ret KILL_PROCESS
   397→///   7. ret ALLOW                       // default: allow
   398→///   8. ret KILL_PROCESS                 // for wrong-arch
   399→/// ```
   400→///
   401→/// The `jt=0, jf=1` pattern keeps jumps short (1 instruction) so we
   402→/// never overflow the 8-bit jt/jf range, regardless of how many
   403→/// syscalls are in each set.
   404→pub fn build_filter() -> Vec<SockFilter> {
   405→    let allowed = allowed_syscalls();
   406→    let trapped = trapped_syscalls();
   407→    let killed  = killed_syscalls();
   408→
   409→    let mut prog: Vec<SockFilter> = Vec::new();
   410→
   411→    // (1) Load arch.
   412→    prog.push(bpf_ld_abs(OFF_ARCH));
   413→    // (2) If arch != EXPECTED, jump to the trailing KILL.
   414→    //     We patch the jf offset after we know the program length.
   415→    let arch_jeq_idx = prog.len();
   416→    prog.push(bpf_jeq(AUDIT_ARCH_EXPECTED, 0, 0)); // patched below
   417→
   418→    // (3) Load syscall number.
   419→    prog.push(bpf_ld_abs(OFF_NR));
   420→
   421→    // (4) Allowed syscalls: jeq + ret ALLOW.
   422→    let mut sorted_allowed: Vec<i32> = allowed.iter().copied()
   423→        .filter(|nr| !trapped.contains(nr) && !killed.contains(nr))
   424→        .collect();
   425→    sorted_allowed.sort_unstable();
   426→    sorted_allowed.dedup();
   427→    for nr in sorted_allowed {
   428→        prog.push(bpf_jeq(nr as u32, 0, 1));  // match: fall to next; else skip 1
   429→        prog.push(bpf_ret(SECCOMP_RET_ALLOW));
   430→    }
   431→
   432→    // (5) Trapped syscalls: jeq + ret TRAP.
   433→    let mut sorted_trapped: Vec<i32> = trapped.iter().copied()
   434→        .filter(|nr| !killed.contains(nr))
   435→        .collect();
   436→    sorted_trapped.sort_unstable();
   437→    sorted_trapped.dedup();
   438→    for nr in sorted_trapped {
   439→        prog.push(bpf_jeq(nr as u32, 0, 1));
   440→        prog.push(bpf_ret(SECCOMP_RET_TRAP));
   441→    }
   442→
   443→    // (6) Killed syscalls: jeq + ret KILL.
   444→    let mut sorted_killed: Vec<i32> = killed.iter().copied().collect();
   445→    sorted_killed.sort_unstable();
   446→    sorted_killed.dedup();
   447→    for nr in sorted_killed {
   448→        prog.push(bpf_jeq(nr as u32, 0, 1));
   449→        prog.push(bpf_ret(SECCOMP_RET_KILL_PROCESS));
   450→    }
   451→
   452→    // (7) Default: allow.
   453→    prog.push(bpf_ret(SECCOMP_RET_ALLOW));
   454→
   455→    // (8) Kill target (for wrong-arch). Patch the arch jeq's jf to
   456→    //     jump here. The jf offset is relative to the *next*
   457→    //     instruction, so it's `kill_idx - arch_jeq_idx - 1`.
   458→    let kill_idx = prog.len();
   459→    prog.push(bpf_ret(SECCOMP_RET_KILL_PROCESS));
   460→    let jf_offset = (kill_idx - arch_jeq_idx - 1) as u8;
   461→    prog[arch_jeq_idx].jf = jf_offset;
   462→
   463→    info!("[KR64][seccomp] BPF filter built: {} instructions", prog.len());
   464→    prog
   465→}

   467→// ============================================================================
   468→// Install the filter + SIGSYS handler.
   469→// ============================================================================
   470→
   471→/// Global mutable state used by the SIGSYS handler. Protected by a
   472→/// Mutex because the handler may run on any thread.
   473→///
   474→/// The handler reads `trapped_syscalls` and `killed_syscalls` to decide
   475→/// what to do with the trapped syscall. We snapshot them at install
   476→/// time into static `OnceLock`s to avoid allocating in the handler
   477→/// (and to avoid the `once_cell` dependency — `OnceLock` is in `std`
   478→/// since Rust 1.70).
   479→static TRAPPED_SET: OnceLock<Mutex<HashSet<i32>>> = OnceLock::new();
   480→static KILLED_SET: OnceLock<Mutex<HashSet<i32>>> = OnceLock::new();
   481→
   482→/// Install the seccomp filter on the calling thread (and all threads
   483→/// in the calling process, via `SECCOMP_FILTER_FLAG_TSYNC`).
   484→///
   485→/// This is the entry point called from `main.rs` after all devices
   486→/// are set up but before the guest init is exec'd.
   487→///
   488→/// Returns `Ok(())` on success, or `Err(errno)` on failure.
   489→pub fn install() -> std::io::Result<()> {
   490→    info!("[KR64][seccomp] installing seccomp filter");
   491→
   492→    // 1. PR_SET_NO_NEW_PRIVS — required before seccomp(2) without
   493→    //    CAP_SYS_ADMIN. Has to be set on every thread; we use the
   494→    //    TSYNC flag below to propagate to all threads in the process.
   495→    let r = unsafe { libc::prctl(PR_SET_NO_NEW_PRIVS, 1, 0, 0, 0) };
   496→    if r != 0 {
   497→        let e = std::io::Error::last_os_error();
   498→        error!("[KR64][seccomp] PR_SET_NO_NEW_PRIVS failed: {}", e);
   499→        return Err(e);
   500→    }
   501→    info!("[KR64][seccomp] PR_SET_NO_NEW_PRIVS set");
   502→
   503→    // 2. Install the SIGSYS handler. We use SA_SIGINFO so we get the
   504→    //    siginfo_t (with si_syscall) and the ucontext_t (so we can
   505→    //    modify the return value and PC).
   506→    install_sigsys_handler()?;
   507→
   508→    // 3. Build the BPF program.
   509→    let prog = build_filter();
   510→
   511→    // 4. seccomp(SECCOMP_SET_MODE_FILTER, TSYNC, &prog).
   512→    //    We use the raw `syscall()` wrapper because libc doesn't wrap
   513→    //    `seccomp(2)` on every target.
   514→    #[repr(C)]
   515→    struct Fprog {
   516→        len:    u16,
   517→        filter: *const SockFilter,
   518→    }
   519→    let fprog = Fprog {
   520→        len:    prog.len() as u16,
   521→        filter: prog.as_ptr(),
   522→    };
   523→
   524→    // seccomp(2) is syscall 277 on aarch64, 317 on x86_64. Use libc's
   525→    // SYS_seccomp constant (defined on all Android targets we support).
   526→    let r = unsafe {
   527→        libc::syscall(
   528→            libc::SYS_seccomp,
   529→            SECCOMP_SET_MODE_FILTER as c_int,
   530→            SECCOMP_FILTER_FLAG_TSYNC as u32,
   531→            &fprog as *const Fprog,
   532→        )
   533→    };
   534→    if r != 0 {
   535→        let e = std::io::Error::last_os_error();
   536→        error!("[KR64][seccomp] SECCOMP_SET_MODE_FILTER failed: {}", e);
   537→        return Err(e);
   538→    }
   539→
   540→    info!("[KR64][seccomp] seccomp filter installed ({} insns, TSYNC)", prog.len());
   541→    Ok(())
   542→}

   544→/// Install the SIGSYS handler that catches `SECCOMP_RET_TRAP` events.
   545→fn install_sigsys_handler() -> std::io::Result<()> {
   546→    let mut sa: sigaction = unsafe { std::mem::zeroed() };
   547→    // sa_sigaction is a union with sa_handler; on Linux we set it to
   548→    // the address of our trampoline. Cast through a raw pointer first
   549→    // to satisfy the `function_casts_as_integer` lint (and to match
   550→    // the `sighandler_t` typedef on glibc).
   551→    sa.sa_sigaction = sigsys_handler as *const () as usize;
   552→    // SA_SIGINFO — pass siginfo + ucontext to the handler.
   553→    // SA_NODEFER  — don't block SIGSYS while in the handler (so a
   554→    //               recursive trap in the handler is visible).
   555→    // SA_RESTART  — restart interrupted syscalls.
   556→    sa.sa_flags = libc::SA_SIGINFO | libc::SA_NODEFER;
   557→    // Don't block any other signals during the handler.
   558→    unsafe { libc::sigemptyset(&mut sa.sa_mask as *mut sigset_t) };
   559→
   560→    let r = unsafe { libc::sigaction(libc::SIGSYS, &sa, std::ptr::null_mut()) };
   561→    if r != 0 {
   562→        let e = std::io::Error::last_os_error();
   563→        error!("[KR64][seccomp] sigaction(SIGSYS) failed: {}", e);
   564→        return Err(e);
   565→    }
   566→    info!("[KR64][seccomp] SIGSYS handler installed at {:p}",
   567→          sigsys_handler as *const c_void);
   568→    Ok(())
   569→}

   571→// ============================================================================
   572→// The SIGSYS handler itself.
   573→// ============================================================================
   574→
   575→/// Kernel-side layout of `siginfo_t` when `si_signo == SIGSYS` (i.e. the
   576→/// `SYS_SECCOMP` / `si_code == 1` path). See Linux's
   577→/// `include/uapi/asm-generic/siginfo.h::__sifields.__sigsys`.
   578→///
   579→/// The libc crate's `siginfo_t` exposes accessors like `si_pid()` /
   580→/// `si_uid()` / `si_addr()` / `si_value()` but does NOT expose
   581→/// `si_syscall`, so we re-interpret the raw bytes through this layout.
   582→/// This is safe because the kernel-userland `siginfo` ABI is fixed and
   583→/// architecture-independent (the `_pad` field after `si_code` exists to
   584→/// align the 64-bit `si_call_addr` on 64-bit platforms; on 32-bit it
   585→/// collapses but is harmless).
   586→#[repr(C)]
   587→struct SigsysSiginfo {
   588→    si_signo:     i32,
   589→    /// For seccomp traps this holds the seccomp return value
   590→    /// (e.g. `SECCOMP_RET_TRAP`).
   591→    si_errno:     i32,
   592→    /// `SYS_SECCOMP` (= 1).
   593→    si_code:      i32,
   594→    /// Alignment padding so `si_call_addr` is 8-byte aligned.
   595→    _pad:         i32,
   596→    /// Address of the syscall instruction that was trapped.
   597→    si_call_addr: *mut c_void,
   598→    /// The trapped syscall number (what we care about).
   599→    si_syscall:   i32,
   600→    /// The audit architecture of the trapped syscall
   601→    /// (e.g. `AUDIT_ARCH_AARCH64`).
   602→    si_arch:      u32,
   603→}
   604→
   605→/// SIGSYS handler — called by the kernel when a `SECCOMP_RET_TRAP` is
   606→/// triggered (i.e. when the guest calls one of the "trapped" syscalls).
   607→///
   608→/// The handler:
   609→///   1. Reads `si_syscall` to learn which syscall was trapped.
   610→///   2. Looks the syscall up in the trapped/killed sets.
   611→///   3. If killed: log and call `_exit(1)` (no return).
   612→///   4. If trapped: emulate the syscall by:
   613→///        a. Setting the return value register to 0 (success) — or
   614→///           `-ENOSYS` if we don't have a real emulation yet.
   615→///        b. Advancing the program counter past the syscall instruction
   616→///           so the kernel doesn't retry it.
   617→///   5. Returns; the kernel restores the (modified) context and
   618→///      continues execution at the new PC.
   619→///
   620→/// # Architecture-specific notes
   621→///
   622→/// On aarch64:
   623→///   - Return value register: `mcontext.regs[0]` (x0)
   624→///   - PC register:           `mcontext.pc`
   625→///   - Syscall instruction:   4 bytes (`svc #0` = `0xd4000001`)
   626→///
   627→/// On x86_64:
   628→///   - Return value register: `mcontext.gregs[REG_RAX]` = `gregs[13]`
   629→///   - PC register:           `mcontext.gregs[REG_RIP]` = `gregs[16]`
   630→///   - Syscall instruction:   2 bytes (`syscall` = `0x0f 0x05`)
   631→extern "C" fn sigsys_handler(_sig: c_int, info: *mut siginfo_t, ctx: *mut c_void) {
   632→    // Reinterpret the siginfo_t pointer as our SIGSYS-specific layout.
   633→    // This is safe because the kernel always uses this layout when
   634→    // delivering SIGSYS from seccomp, regardless of how the libc crate
   635→    // chooses to expose the union.
   636→    let si = unsafe { &*(info as *const SigsysSiginfo) };
   637→    let uc = unsafe { &mut *(ctx as *mut ucontext_t) };
   638→
   639→    // Read the syscall number.
   640→    let syscall_nr = si.si_syscall;
   641→
   642→    // Decide what to do.
   643→    let action = classify(syscall_nr);
   644→
   645→    match action {
   646→        Action::Kill => {
   647→            error!(
   648→                "[KR64][seccomp] BLOCKED.SYSCALL.FAILED: killed guest for syscall {}",
   649→                syscall_nr
   650→            );
   651→            unsafe { libc::_exit(1) };
   652→        }
   653→        Action::Emulate { retval } => {
   654→            // Emulate: set return value, advance PC past syscall instr.
   655→            warning!(
   656→                "[KR64][seccomp] BLOCKED.SYSCALL.FAILED: trapped syscall {} → emulated (retval={})",
   657→                syscall_nr, retval
   658→            );
   659→            set_return_value(uc, retval);
   660→            advance_pc(uc);
   661→        }
   662→        Action::Passthrough => {
   663→            // This shouldn't happen — only trapped syscalls reach the
   664→            // handler. But if it does, log and let the kernel retry
   665→            // (don't advance PC) so the syscall actually executes.
   666→            warning!(
   667→                "[KR64][seccomp] SIGSYS for non-trapped syscall {} — passthrough",
   668→                syscall_nr
   669→            );
   670→        }
   671→    }
   672→}
   673→
   674→#[derive(Debug, Clone, Copy)]
   675→enum Action {
   676→    Kill,
   677→    Emulate { retval: i64 },
   678→    Passthrough,
   679→}
   680→
   681→/// Look the syscall up in the trapped/killed tables and decide what to
   682→/// do. The mapping is:
   683→///   - killed set → Action::Kill
   684→///   - trapped set → Action::Emulate { retval: 0 } (or syscall-specific)
   685→///   - anything else → Action::Passthrough (shouldn't happen)
   686→fn classify(syscall_nr: i32) -> Action {
   687→    // `get_or_init` snapshots the syscall sets on first use. This is
   688→    // safe to call from the SIGSYS handler because OnceLock's init path
   689→    // is race-free (the OS guarantees atomic initialisation).
   690→    let killed = KILLED_SET.get_or_init(|| Mutex::new(killed_syscalls()));
   691→    if let Ok(killed) = killed.lock() {
   692→        if killed.contains(&syscall_nr) {
   693→            return Action::Kill;
   694→        }
   695→    }
   696→    let trapped = TRAPPED_SET.get_or_init(|| Mutex::new(trapped_syscalls()));
   697→    if let Ok(trapped) = trapped.lock() {
   698→        if trapped.contains(&syscall_nr) {
   699→            // Per-syscall emulation. For now everything trapped is
   700→            // emulated as success (retval = 0) — the production
   701→            // version will dispatch to per-syscall handlers in
   702→            // mount_mgr / netlink / etc.
   703→            return Action::Emulate { retval: emulate_syscall(syscall_nr) };
   704→        }
   705→    }
   706→    Action::Passthrough
   707→}
   708→
   709→/// Per-syscall emulation — return a fake return value.
   710→///
   711→/// For the MVP, every trapped syscall returns 0 (success). The
   712→/// production version will:
   713→///   - `mount`   → call `mount_mgr::handle_mount(args…)`, return its result
   714→///   - `umount2` → call `mount_mgr::handle_umount(args…)`, return its result
   715→///   - `swapon`  → return 0 (no-op)
   716→///   - `swapoff` → return 0 (no-op)
   717→///   - `acct`    → return 0 (no-op)
   718→///   - `reboot`  → return -EPERM (we don't reboot the host!)
   719→fn emulate_syscall(_syscall_nr: i32) -> i64 {
   720→    // TODO: dispatch to per-syscall handlers (mount_mgr, etc.).
   721→    0
   722→}
   723→
   724→/// Set the syscall return value in the saved context.
   725→///
   726→/// On aarch64: x0 (regs[0]) is the return value register.
   727→/// On x86_64: rax (gregs[REG_RAX]) is the return value register.
   728→#[cfg(target_arch = "aarch64")]
   729→fn set_return_value(uc: &mut ucontext_t, retval: i64) {
   730→    uc.uc_mcontext.regs[0] = retval as u64;
   731→}
   732→
   733→#[cfg(target_arch = "x86_64")]
   734→fn set_return_value(uc: &mut ucontext_t, retval: i64) {
   735→    // REG_RAX = 13 (from <sys/ucontext.h> on Linux x86_64).
   736→    const REG_RAX: usize = 13;
   737→    uc.uc_mcontext.gregs[REG_RAX] = retval;
   738→}
   739→
   740→/// Advance the program counter past the syscall instruction.
   741→///
   742→/// On aarch64: every instruction is 4 bytes (svc #0 = 0xd4000001).
   743→/// On x86_64:  the syscall instruction is 2 bytes (0x0f 0x05).
   744→#[cfg(target_arch = "aarch64")]
   745→fn advance_pc(uc: &mut ucontext_t) {
   746→    uc.uc_mcontext.pc = uc.uc_mcontext.pc.wrapping_add(4);
   747→}
   748→
   749→#[cfg(target_arch = "x86_64")]
   750→fn advance_pc(uc: &mut ucontext_t) {
   751→    // REG_RIP = 16 (from <sys/ucontext.h> on Linux x86_64).
   752→    const REG_RIP: usize = 16;
   753→    let rip = uc.uc_mcontext.gregs[REG_RIP];
   754→    uc.uc_mcontext.gregs[REG_RIP] = rip.wrapping_add(2);
   755→}
