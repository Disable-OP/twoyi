    12→//! Seccomp filter + SIGSYS handler — the "kernel replacement" syscall
    13→//! interception layer.
    14→//!
    15→//! This mirrors what VM's `libkr64.so` does (see `VM_KR64_ANALYSIS.md`
    16→//! §12 and §11):
    17→//!
    18→//! 1. `prctl(PR_SET_NO_NEW_PRIVS, 1)` — required before installing a
    19→//!    seccomp filter without `CAP_SYS_ADMIN`.
    20→//! 2. `sigaction(SIGSYS, …)` — install a handler that catches trapped
    21→//!    syscalls. The string `__NR_rt_sigaction SIGSYS` was decoded from
    22→//!    `libkr64.11.so`'s `.data` (key 0x1a).
    23→//! 3. `seccomp(SECCOMP_SET_MODE_FILTER, …)` — install a BPF program
    24→//!    that traps forbidden syscalls. The BPF program returns
    25→//!    `SECCOMP_RET_TRAP` for "intercepted" syscalls (mount, umount,
    26→//!    reboot, kexec_load, init_module, etc.) and `SECCOMP_RET_ALLOW`
    27→//!    for everything else.
    28→//! 4. The SIGSYS handler logs `BLOCKED.SYSCALL.FAILED: <nr>` (string
    29→//!    decoded from libkr64.so with key 0xc9 / 0xc2) and either:
    30→//!    - Emulates the syscall (sets the return value to 0 / fake-success
    31→//!      and advances the PC past the syscall instruction), OR
    32→//!    - Kills the guest (via `_exit`).
    33→//!
    34→//! # Why `SECCOMP_RET_TRAP`, not `SECCOMP_RET_TRACE`
    35→//!
    36→//! The task spec mentions `SECCOMP_RET_TRACE`, but `SECCOMP_RET_TRACE`
    37→//! requires a ptrace tracer to be attached to the process — without
    38→//! one, the kernel falls back to `SECCOMP_RET_KILL`. The mechanism
    39→//! that actually delivers `SIGSYS` (and the one VM uses, per the
    40→//! analysis) is `SECCOMP_RET_TRAP`. We expose `SECCOMP_RET_TRACE` as
    41→//! a constant for callers that want to switch to ptrace-mode tracing
    42→//! later (e.g. for debugging the guest), but the default filter uses
    43→//! `SECCOMP_RET_TRAP`.
    44→//!
    45→//! # Why we don't trap EVERY syscall
    46→//!
    47→//! VM's filter traps a small set of dangerous syscalls (mount, reboot,
    48→//! kexec, init_module, ptrace, etc.) and lets everything else go
    49→//! through. This is the right tradeoff — trapping every syscall (and
    50→//! emulating each one in the handler) would be 100× slower and would
    51→//! require a complete kernel emulation layer. We only trap syscalls
    52→//! that:
    53→//!   (a) would let the guest escape its sandbox, OR
    54→//!   (b) need to be redirected to a userspace implementation
    55→//!       (mount → bind mount, umount → unbind, swapon → no-op, etc.)

   123→/// A single BPF instruction. Layout matches `struct sock_filter`:

   129→    pub jt:   u8,
   130→    pub jf:   u8,
   131→    pub k:    u32,

   134→/// Build a `BPF_LD | BPF_W | BPF_ABS` instruction — load a 32-bit word
   135→/// from offset `k` of `struct seccomp_data` into the accumulator.
   136→const fn bpf_ld_abs(k: u32) -> SockFilter {
   137→    SockFilter { code: BPF_LD | BPF_W | BPF_ABS, jt: 0, jf: 0, k }
   138→}
   139→
   140→/// Build a `BPF_JMP | BPF_JEQ | BPF_K` instruction — if accumulator
   141→/// equals `k`, jump `jt` instructions forward, else jump `jf`.
   142→const fn bpf_jeq(k: u32, jt: u8, jf: u8) -> SockFilter {
   143→    SockFilter { code: BPF_JMP | BPF_JEQ | BPF_K, jt, jf, k }
   144→}

   146→/// Build a `BPF_RET | BPF_K` instruction — return `k` from the filter.
   147→const fn bpf_ret(k: u32) -> SockFilter {
   148→    SockFilter { code: BPF_RET | BPF_K, jt: 0, jf: 0, k }

   375→// Build the BPF program.

   378→/// Compile the allow/trap/kill classification into a BPF program.
   379→///
   380→/// The program structure is:
   381→/// ```text
   382→///   1. ld arch                         // load audit arch
   383→///   2. jeq EXPECTED_ARCH, jt=0, jf=N   // if wrong arch, jump to kill
   384→///   3. ld nr                           // load syscall number
   385→///   4. for each allowed syscall:
   386→///        jeq nr, jt=0, jf=1            // if match: fall to ret ALLOW
   387→///        ret ALLOW                      // else: skip to next jeq
   388→///   5. for each trapped syscall:
   389→///        jeq nr, jt=0, jf=1
   390→///        ret TRAP
   391→///   6. for each killed syscall:
   392→///        jeq nr, jt=0, jf=1
   393→///        ret KILL_PROCESS
   394→///   7. ret ALLOW                       // default: allow
   395→///   8. ret KILL_PROCESS                 // for wrong-arch
   396→/// ```
   397→///
   398→/// The `jt=0, jf=1` pattern keeps jumps short (1 instruction) so we
   399→/// never overflow the 8-bit jt/jf range, regardless of how many
   400→/// syscalls are in each set.
   401→pub fn build_filter() -> Vec<SockFilter> {
   402→    let allowed = allowed_syscalls();
   403→    let trapped = trapped_syscalls();
   404→    let killed  = killed_syscalls();
   405→
   406→    let mut prog: Vec<SockFilter> = Vec::new();
   407→
   408→    // (1) Load arch.
   409→    prog.push(bpf_ld_abs(OFF_ARCH));
   410→    // (2) If arch != EXPECTED, jump to the trailing KILL.
   411→    //     We patch the jf offset after we know the program length.
   412→    let arch_jeq_idx = prog.len();
   413→    prog.push(bpf_jeq(AUDIT_ARCH_EXPECTED, 0, 0)); // patched below
   414→
   415→    // (3) Load syscall number.
   416→    prog.push(bpf_ld_abs(OFF_NR));
   417→
   418→    // (4) Allowed syscalls: jeq + ret ALLOW.
   419→    let mut sorted_allowed: Vec<i32> = allowed.iter().copied()
   420→        .filter(|nr| !trapped.contains(nr) && !killed.contains(nr))
   421→        .collect();
   422→    sorted_allowed.sort_unstable();
   423→    sorted_allowed.dedup();
   424→    for nr in sorted_allowed {
   425→        prog.push(bpf_jeq(nr as u32, 0, 1));  // match: fall to next; else skip 1
   426→        prog.push(bpf_ret(SECCOMP_RET_ALLOW));
   427→    }
   428→
   429→    // (5) Trapped syscalls: jeq + ret TRAP.
   430→    let mut sorted_trapped: Vec<i32> = trapped.iter().copied()
   431→        .filter(|nr| !killed.contains(nr))
   432→        .collect();
   433→    sorted_trapped.sort_unstable();
   434→    sorted_trapped.dedup();
   435→    for nr in sorted_trapped {
   436→        prog.push(bpf_jeq(nr as u32, 0, 1));
   437→        prog.push(bpf_ret(SECCOMP_RET_TRAP));
   438→    }
   439→
   440→    // (6) Killed syscalls: jeq + ret KILL.
   441→    let mut sorted_killed: Vec<i32> = killed.iter().copied().collect();
   442→    sorted_killed.sort_unstable();
   443→    sorted_killed.dedup();
   444→    for nr in sorted_killed {
   445→        prog.push(bpf_jeq(nr as u32, 0, 1));
   446→        prog.push(bpf_ret(SECCOMP_RET_KILL_PROCESS));
   447→    }
   448→
   449→    // (7) Default: allow.
   450→    prog.push(bpf_ret(SECCOMP_RET_ALLOW));
   451→
   452→    // (8) Kill target (for wrong-arch). Patch the arch jeq's jf to
   453→    //     jump here. The jf offset is relative to the *next*
   454→    //     instruction, so it's `kill_idx - arch_jeq_idx - 1`.
   455→    let kill_idx = prog.len();
   456→    prog.push(bpf_ret(SECCOMP_RET_KILL_PROCESS));
   457→    let jf_offset = (kill_idx - arch_jeq_idx - 1) as u8;
   458→    prog[arch_jeq_idx].jf = jf_offset;
   459→
   460→    info!("[KR64][seccomp] BPF filter built: {} instructions", prog.len());
   461→    prog
   462→}

   468→/// Global mutable state used by the SIGSYS handler. Protected by a
   469→/// Mutex because the handler may run on any thread.
   470→///
   471→/// The handler reads `trapped_syscalls` and `killed_syscalls` to decide
   472→/// what to do with the trapped syscall. We snapshot them at install
   473→/// time into static Lazy to avoid allocating in the handler.
   474→static TRAPPED_SET: Lazy<Mutex<HashSet<i32>>> = Lazy::new(|| {
   475→    Mutex::new(trapped_syscalls())
   476→});
   477→static KILLED_SET: Lazy<Mutex<HashSet<i32>>> = Lazy::new(|| {
   478→    Mutex::new(killed_syscalls())
   479→});

   481→/// Install the seccomp filter on the calling thread (and all threads
   482→/// in the calling process, via `SECCOMP_FILTER_FLAG_TSYNC`).
   483→///
   484→/// This is the entry point called from `main.rs` after all devices
   485→/// are set up but before the guest init is exec'd.
   486→///
   487→/// Returns `Ok(())` on success, or `Err(errno)` on failure.
   488→pub fn install() -> std::io::Result<()> {
   489→    info!("[KR64][seccomp] installing seccomp filter");
   490→
   491→    // 1. PR_SET_NO_NEW_PRIVS — required before seccomp(2) without
   492→    //    CAP_SYS_ADMIN. Has to be set on every thread; we use the
   493→    //    TSYNC flag below to propagate to all threads in the process.
   494→    let r = unsafe { libc::prctl(PR_SET_NO_NEW_PRIVS, 1, 0, 0, 0) };
   495→    if r != 0 {
   496→        let e = std::io::Error::last_os_error();
   497→        error!("[KR64][seccomp] PR_SET_NO_NEW_PRIVS failed: {}", e);
   498→        return Err(e);
   499→    }
   500→    info!("[KR64][seccomp] PR_SET_NO_NEW_PRIVS set");
   501→
   502→    // 2. Install the SIGSYS handler. We use SA_SIGINFO so we get the
   503→    //    siginfo_t (with si_syscall) and the ucontext_t (so we can
   504→    //    modify the return value and PC).
   505→    install_sigsys_handler()?;
   506→
   507→    // 3. Build the BPF program.
   508→    let prog = build_filter();
   509→
   510→    // 4. seccomp(SECCOMP_SET_MODE_FILTER, TSYNC, &prog).
   511→    //    We use the raw `syscall()` wrapper because libc doesn't wrap
   512→    //    `seccomp(2)` on every target.
   513→    #[repr(C)]
   514→    struct Fprog {
   515→        len:    u16,
   516→        filter: *const SockFilter,
   517→    }
   518→    let fprog = Fprog {
   519→        len:    prog.len() as u16,
   520→        filter: prog.as_ptr(),
   521→    };
   522→
   523→    // seccomp(2) is syscall 277 on aarch64, 317 on x86_64. Use libc's
   524→    // SYS_seccomp constant (defined on all Android targets we support).
   525→    let r = unsafe {
   526→        libc::syscall(
   527→            libc::SYS_seccomp,
   528→            SECCOMP_SET_MODE_FILTER as c_int,
   529→            SECCOMP_FILTER_FLAG_TSYNC as u32,
   530→            &fprog as *const Fprog,
   531→        )
   532→    };
   533→    if r != 0 {
   534→        let e = std::io::Error::last_os_error();
   535→        error!("[KR64][seccomp] SECCOMP_SET_MODE_FILTER failed: {}", e);
   536→        return Err(e);
   537→    }
   538→
   539→    info!("[KR64][seccomp] seccomp filter installed ({} insns, TSYNC)", prog.len());
   540→    Ok(())
   541→}

   543→/// Install the SIGSYS handler that catches `SECCOMP_RET_TRAP` events.
   544→fn install_sigsys_handler() -> std::io::Result<()> {
   545→    let mut sa: sigaction = unsafe { std::mem::zeroed() };
   546→    // sa_sigaction is a union with sa_handler; on Linux we set it to
   547→    // the address of our trampoline. Rust's libc exposes it as a
   548→    // `usize` (sighandler_t).
   549→    sa.sa_sigaction = sigsys_handler as usize;
   550→    // SA_SIGINFO — pass siginfo + ucontext to the handler.
   551→    // SA_NODEFER  — don't block SIGSYS while in the handler (so a
   552→    //               recursive trap in the handler is visible).
   553→    // SA_RESTART  — restart interrupted syscalls.
   554→    sa.sa_flags = libc::SA_SIGINFO | libc::SA_NODEFER;
   555→    // Don't block any other signals during the handler.
   556→    unsafe { libc::sigemptyset(&mut sa.sa_mask as *mut sigset_t) };
   557→
   558→    let r = unsafe { libc::sigaction(libc::SIGSYS, &sa, std::ptr::null_mut()) };
   559→    if r != 0 {
   560→        let e = std::io::Error::last_os_error();
   561→        error!("[KR64][seccomp] sigaction(SIGSYS) failed: {}", e);
   562→        return Err(e);
   563→    }
   564→    info!("[KR64][seccomp] SIGSYS handler installed at {:p}",
   565→          sigsys_handler as *const c_void);
   566→    Ok(())
   567→}

   569→// ============================================================================
   570→// The SIGSYS handler itself.
   571→// ============================================================================
   572→
   573→/// SIGSYS handler — called by the kernel when a `SECCOMP_RET_TRAP` is
   574→/// triggered (i.e. when the guest calls one of the "trapped" syscalls).
   575→///
   576→/// The handler:
   577→///   1. Reads `si_syscall` to learn which syscall was trapped.
   578→///   2. Looks the syscall up in the trapped/killed sets.
   579→///   3. If killed: log and call `_exit(1)` (no return).
   580→///   4. If trapped: emulate the syscall by:
   581→///        a. Setting the return value register to 0 (success) — or
   582→///           `-ENOSYS` if we don't have a real emulation yet.
   583→///        b. Advancing the program counter past the syscall instruction
   584→///           so the kernel doesn't retry it.
   585→///   5. Returns; the kernel restores the (modified) context and
   586→///      continues execution at the new PC.
   587→///
   588→/// # Architecture-specific notes
   589→///
   590→/// On aarch64:
   591→///   - Return value register: `mcontext.regs[0]` (x0)
   592→///   - PC register:           `mcontext.pc`
   593→///   - Syscall instruction:   4 bytes (`svc #0` = `0xd4000001`)
   594→///
   595→/// On x86_64:
   596→///   - Return value register: `mcontext.gregs[REG_RAX]` = `gregs[13]`
   597→///   - PC register:           `mcontext.gregs[REG_RIP]` = `gregs[16]`
   598→///   - Syscall instruction:   2 bytes (`syscall` = `0x0f 0x05`)
   599→extern "C" fn sigsys_handler(_sig: c_int, info: *mut siginfo_t, ctx: *mut c_void) {
   600→    let si = unsafe { &*info };
   601→    let uc = unsafe { &mut *(ctx as *mut ucontext_t) };
   602→
   603→    // Read the syscall number from siginfo.
   604→    let syscall_nr = unsafe { si.si_syscall() };
   605→
   606→    // Decide what to do.
   607→    let action = classify(syscall_nr);
   608→
   609→    match action {
   610→        Action::Kill => {
   611→            error!(
   612→                "[KR64][seccomp] BLOCKED.SYSCALL.FAILED: killed guest for syscall {}",
   613→                syscall_nr
   614→            );
   615→            unsafe { libc::_exit(1) };
   616→        }
   617→        Action::Emulate { retval } => {
   618→            // Emulate: set return value, advance PC past syscall instr.
   619→            warn!(
   620→                "[KR64][seccomp] BLOCKED.SYSCALL.FAILED: trapped syscall {} → emulated (retval={})",
   621→                syscall_nr, retval
   622→            );
   623→            set_return_value(uc, retval);
   624→            advance_pc(uc);
   625→        }
   626→        Action::Passthrough => {
   627→            // This shouldn't happen — only trapped syscalls reach the
   628→            // handler. But if it does, log and let the kernel retry
   629→            // (don't advance PC) so the syscall actually executes.
   630→            warn!(
   631→                "[KR64][seccomp] SIGSYS for non-trapped syscall {} — passthrough",
   632→                syscall_nr
   633→            );
   634→        }
   635→    }
   636→}
   637→
   638→#[derive(Debug, Clone, Copy)]
   639→enum Action {
   640→    Kill,
   641→    Emulate { retval: i64 },
   642→    Passthrough,
   643→}
   644→
   645→/// Look the syscall up in the trapped/killed tables and decide what to
   646→/// do. The mapping is:
   647→///   - killed set → Action::Kill
   648→///   - trapped set → Action::Emulate { retval: 0 } (or syscall-specific)
   649→///   - anything else → Action::Passthrough (shouldn't happen)
   650→fn classify(syscall_nr: i32) -> Action {
   651→    if let Ok(killed) = KILLED_SET.lock() {
   652→        if killed.contains(&syscall_nr) {
   653→            return Action::Kill;
   654→        }
   655→    }
   656→    if let Ok(trapped) = TRAPPED_SET.lock() {
   657→        if trapped.contains(&syscall_nr) {
   658→            // Per-syscall emulation. For now everything trapped is
   659→            // emulated as success (retval = 0) — the production
   660→            // version will dispatch to per-syscall handlers in
   661→            // mount_mgr / netlink / etc.
   662→            return Action::Emulate { retval: emulate_syscall(syscall_nr) };
   663→        }
   664→    }
   665→    Action::Passthrough
   666→}
   667→
   668→/// Per-syscall emulation — return a fake return value.
   669→///
   670→/// For the MVP, every trapped syscall returns 0 (success). The
   671→/// production version will:
   672→///   - `mount`   → call `mount_mgr::handle_mount(args…)`, return its result
   673→///   - `umount2` → call `mount_mgr::handle_umount(args…)`, return its result
   674→///   - `swapon`  → return 0 (no-op)
   675→///   - `swapoff` → return 0 (no-op)
   676→///   - `acct`    → return 0 (no-op)
   677→///   - `reboot`  → return -EPERM (we don't reboot the host!)
   678→fn emulate_syscall(_syscall_nr: i32) -> i64 {
   679→    // TODO: dispatch to per-syscall handlers (mount_mgr, etc.).
   680→    0
   681→}
   682→
   683→/// Set the syscall return value in the saved context.
   684→///
   685→/// On aarch64: x0 (regs[0]) is the return value register.
   686→/// On x86_64: rax (gregs[REG_RAX]) is the return value register.
   687→#[cfg(target_arch = "aarch64")]
   688→fn set_return_value(uc: &mut ucontext_t, retval: i64) {
   689→    uc.uc_mcontext.regs[0] = retval as u64;
   690→}
   691→
   692→#[cfg(target_arch = "x86_64")]
   693→fn set_return_value(uc: &mut ucontext_t, retval: i64) {
   694→    // REG_RAX = 13 (from <sys/ucontext.h> on Linux x86_64).
   695→    const REG_RAX: usize = 13;
   696→    uc.uc_mcontext.gregs[REG_RAX] = retval;
   697→}
   698→
   699→/// Advance the program counter past the syscall instruction.
   700→///
   701→/// On aarch64: every instruction is 4 bytes (svc #0 = 0xd4000001).
   702→/// On x86_64:  the syscall instruction is 2 bytes (0x0f 0x05).
   703→#[cfg(target_arch = "aarch64")]
   704→fn advance_pc(uc: &mut ucontext_t) {
   705→    uc.uc_mcontext.pc = uc.uc_mcontext.pc.wrapping_add(4);
   706→}
   707→
   708→#[cfg(target_arch = "x86_64")]
   709→fn advance_pc(uc: &mut ucontext_t) {
   710→    // REG_RIP = 16 (from <sys/ucontext.h> on Linux x86_64).
   711→    const REG_RIP: usize = 16;
   712→    let rip = uc.uc_mcontext.gregs[REG_RIP];
   713→    uc.uc_mcontext.gregs[REG_RIP] = rip.wrapping_add(2);
   714→}
