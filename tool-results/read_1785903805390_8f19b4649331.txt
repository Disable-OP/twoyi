    12→//! Twoyi kernel-replacement daemon — the Rust port of VM's `libkr64.so`.
    13→//!
    14→//! See `VM_KR64_ANALYSIS.md` and `GSI_BOOT_PLAN.md` for the full
    15→//! background. The short version: twoyi needs a userspace daemon that
    16→//! materialises a virtual `/dev` tree for the guest Android, installs
    17→//! a seccomp filter that traps "dangerous" syscalls, emulates `/proc`
    18→//! and `/sys`, manages the per-VM mount namespace, and exec's the
    19→//! guest's `/system/bin/init`. This crate is that daemon.
    20→//!
    21→//! # Build
    22→//!
    23→//! The crate builds as BOTH a cdylib (`libkr64.so` — directly
    24→//! executable via the PIE hack) AND a regular binary (`kr64`). See
    25→//! `Cargo.toml` and `build.rs` for details.
    26→//!
    27→//! # Usage
    28→//!
    29→//! ```sh
    30→//! # Direct execution via the PIE-hacked cdylib:
    31→//! ./libkr64.so --rootfs /data/.../fs --data-dir /data/... --vmid 0
    32→//!
    33→//! # Or via the regular binary:
    34→//! ./kr64 --rootfs /data/.../fs --data-dir /data/... --vmid 0
    35→//! ```
    36→//!
    37→//! # Module layout
    38→//!
    39→//! * [`devices`]    — virtual `/dev` device creation (qemu_pipe,
    40→//!                    touch, key, event, gb, gb2).
    41→//! * [`binder`]     — per-VM `/vm{id}/dev/binder` Unix socket + binder
    42→//!                    transaction proxy (skeleton; see
    43→//!                    `download/BINDER_SKELETON.md`).
    44→//! * [`audio`]      — virtual `/dev/audio` Unix socket + bidirectional
    45→//!                    PCM pump (playback + capture skeleton; see
    46→//!                    `download/AUDIO_SENSOR_HAL.md`).
    47→//! * [`sensors`]    — virtual `/dev/sensors` Unix socket + multiplexed
    48→//!                    12-sensor HAL (accel/mag/gyro/... skeleton; see
    49→//!                    `download/AUDIO_SENSOR_HAL.md`).
    50→//! * [`seccomp`]    — BPF seccomp filter + SIGSYS handler.
    51→//! * [`proc_emu`]   — synthesised `/proc` tree (version, cpuinfo,

    53→//! * [`mount_mgr`]  — `unshare(CLONE_NEWNS)` + `pivot_root` + tmpfs

   291→/// Run the daemon. Returns the exit code.
   292→///
   293→/// This is the shared entry point called by both:
   294→///   * `kr64_main` (the cdylib's PIE entry point, see below), and
   295→///   * `main()` in `src/main.rs` (the bin target).
   296→///
   297→/// Steps:
   298→///   1. Parse args.
   299→///   2. Create all virtual `/dev` devices (qemu_pipe, touch, key,
   300→///      event, gb, gb2).
   301→///   3. Populate `/proc` (version, cpuinfo, meminfo, cmdline, …).
   302→///   4. Set up mount namespace (unshare + bind mounts + tmpfs mounts).
   303→///   5. Fork:
   304→///      - Child: pivot_root → install seccomp → exec /system/bin/init.
   305→///      - Parent: run the device-accept loop (spawns one thread per
   306→///        device socket; for the MVP each thread just accepts and
   307→///        echoes).
   308→///   6. Wait for the child (the guest init) to exit; propagate its
   309→///      exit code.
   310→pub fn run<I: IntoIterator<Item = String>>(args: I) -> i32 {
   311→    let cfg = match parse_args(args) {
   312→        Ok(c) => c,
   313→        Err(e) => {
   314→            // If --help, print to stdout and exit 0; otherwise stderr
   315→            // and exit 2.
   316→            if e.starts_with("twoyi kr64") {
   317→                println!("{}", e);
   318→                return 0;
   319→            }
   320→            eprintln!("kr64: {}", e);
   321→            return 2;
   322→        }
   323→    };
   324→
   325→    info!("[KR64] starting daemon with config: {:?}", cfg);
   326→
   327→    // ---------------------------------------------------------------
   328→    // Step 2: create virtual /dev devices.
   329→    // ---------------------------------------------------------------
   330→    let device_set = match devices::create_all_devices(&cfg.rootfs, &cfg.data_dir) {
   331→        Ok(d) => d,
   332→        Err(e) => {
   333→            error!("[KR64] failed to create devices: {}", e);
   334→            return 1;
   335→        }
   336→    };
   337→
   338→    // Create the marker files the guest init looks for.
   339→    if let Err(e) = devices::create_coldboot_done_marker(&cfg.rootfs) {
   340→        warning!("[KR64] failed to create .coldboot_done marker: {}", e);
   341→    }
   342→    if let Err(e) = devices::create_busybox_marker(&cfg.rootfs) {
   343→        warning!("[KR64] failed to create .busybox marker: {}", e);
   344→    }
   345→
   346→    // ---------------------------------------------------------------
   347→    // Step 2.5: create the per-VM binder device + spawn the binder
   348→    // proxy. See `download/BINDER_SKELETON.md` and
   349→    // `app/rs/kr64/src/binder.rs` for the full story. The short
   350→    // version: this creates `{rootfs}/vm{id}/dev/binder` as a Unix
   351→    // socket + a `{rootfs}/dev/binder` symlink, and spawns an
   352→    // accept-thread + worker-pool that dispatches BINDER_* ioctls
   353→    // from the guest. The handle is held for the lifetime of `run()`
   354→    // so the proxy is shut down when the guest exits.
   355→    // ---------------------------------------------------------------
   356→    let _binder_handle = match binder::create_binder_device(&cfg.rootfs, cfg.vmid)
   357→        .and_then(|path| binder::BinderProxy::new(cfg.vmid, &path))
   358→        .and_then(|proxy| proxy.spawn())
   359→    {
   360→        Ok(h) => {
   361→            info!("[KR64] binder proxy listening at {} (vm{})", h.path(), cfg.vmid);
   362→            Some(h)
   363→        }
   364→        Err(e) => {
   365→            // Non-fatal: the guest can still boot against the host's
   366→            // /dev/binder if it's bind-mounted in (the current twoyi
   367→            // approach). The binder proxy is only needed for full
   368→            // service-manager virtualisation.
   369→            warning!("[KR64] failed to start binder proxy: {} — falling back to host binder", e);
   370→            None
   371→        }
   372→    };
   373→
   374→    // ---------------------------------------------------------------
   375→    // Step 2.6: create the audio device + spawn the accept/pump
   376→    // thread. See `download/AUDIO_SENSOR_HAL.md` and
   377→    // `app/rs/kr64/src/audio.rs` for the full story. The short
   378→    // version: this creates `{rootfs}/dev/audio` as a Unix socket
   379→    // and spawns an accept-thread + worker-pool that reads a
   380→    // 16-byte header per connection and pumps raw PCM in both
   381→    // directions. The actual AudioTrack/AudioRecord integration is
   382→    // stubbed (no JNI yet) — the pump compiles and exercises the
   383→    // protocol but produces no sound until the Java side is wired
   384→    // up in a follow-up task.
   385→    // ---------------------------------------------------------------
   386→    let _audio_handle = match audio::create_audio_device(&cfg.rootfs)
   387→        .and_then(|dev| dev.spawn())
   388→    {
   389→        Ok(h) => {
   390→            info!("[KR64] audio device listening at {}", h.path());
   391→            Some(h)
   392→        }
   393→        Err(e) => {
   394→            // Non-fatal: the guest can still boot without sound —
   395→            // AudioFlinger's connect() to /dev/audio will fail and
   396→            // the guest's audio HAL will fall back to silence / a
   397→            // null output. Sound is the user's primary use case
   398→            // though, so this warning is worth surfacing.
   399→            warning!("[KR64] failed to start audio device: {} — guest will have no sound", e);
   400→            None
   401→        }
   402→    };
   403→
   404→    // ---------------------------------------------------------------
   405→    // Step 2.7: create the sensor device + spawn the accept/pump
   406→    // thread. See `download/AUDIO_SENSOR_HAL.md` and
   407→    // `app/rs/kr64/src/sensors.rs` for the full story. The short
   408→    // version: this creates `{rootfs}/dev/sensors` as a Unix socket
   409→    // and spawns an accept-thread + worker-pool that reads 12-byte
   410→    // control requests (ENABLE/DISABLE/CHECK_SUPPORT/SET_DELAY) and
   411→    // pushes 24-byte SensorEvent records to the guest when sensors
   412→    // are enabled. The actual SensorManager integration is stubbed
   413→    // (no JNI yet) — the control loop replies false to every
   414→    // CHECK_SUPPORT and the pump never produces events. This is the
   415→    // deliberate "skeleton" boundary, mirroring audio.rs.
   416→    // ---------------------------------------------------------------
   417→    let _sensor_handle = match sensors::create_sensor_device(&cfg.rootfs)
   418→        .and_then(|dev| dev.spawn())
   419→    {
   420→        Ok(h) => {
   421→            info!("[KR64] sensor device listening at {}", h.path());
   422→            Some(h)
   423→        }
   424→        Err(e) => {
   425→            // Non-fatal: the guest can still boot without sensors —
   426→            // the guest's sensor HAL will see "no sensors available"
   427→            // and `SensorManager.getDefaultSensor()` will return null.
   428→            // Apps that hard-require a sensor (e.g. compass apps)
   429→            // will crash, but the boot proceeds.
   430→            warning!("[KR64] failed to start sensor device: {} — guest will have no sensors", e);
   431→            None
   432→        }
   433→    };
   434→
   435→    // ---------------------------------------------------------------
   436→    // Step 3: populate /proc.
   437→    // ---------------------------------------------------------------
   438→    // We synthesise for an 8-core, 4 GB guest by default. A production
   439→    // version would query the host's CPU count and memory size and
   440→    // scale these accordingly.
   441→    let cpu_count = 8u32;
   442→    let mem_mb = 4096u64;
   443→    if let Err(e) = proc_emu::populate_proc(&cfg.rootfs, cpu_count, mem_mb) {
   444→        warning!("[KR64] failed to populate /proc: {}", e);
   445→    }
   446→
   447→    // ---------------------------------------------------------------
   448→    // Step 4: set up mount namespace + bind mounts + tmpfs.
   449→    // ---------------------------------------------------------------
   450→    // NOTE: We do NOT do this yet — the mount setup has to happen
   451→    // inside the forked child, because once we pivot_root we lose
   452→    // access to the host's paths (and the parent needs to keep
   453→    // accepting connections on the device sockets it bound above).
   454→    //
   455→    // The flow is:
   456→    //   parent: bind device sockets (already done above)
   457→    //   parent: fork()
   458→    //     child: mount_mgr::setup_mounts(cfg)
   459→    //     child: seccomp::install()
   460→    //     child: execve(/system/bin/init)
   461→    //   parent: accept loop on device sockets
   462→
   463→    // ---------------------------------------------------------------
   464→    // Step 5: fork + exec the guest.
   465→    // ---------------------------------------------------------------
   466→    info!("[KR64] forking guest process");
   467→    let pid = unsafe { libc::fork() };
   468→    if pid < 0 {
   469→        let e = std::io::Error::last_os_error();
   470→        error!("[KR64] fork failed: {}", e);
   471→        return 1;
   472→    }
   473→
   474→    if pid == 0 {
   475→        // ----- CHILD (will become the guest init) -----
   476→        info!("[KR64][child] setting up mounts");
   477→        let mount_cfg = mount_mgr::MountConfig {
   478→            rootfs:           cfg.rootfs.clone(),
   479→            rom_dir:          cfg.rom_dir.clone(),
   480→            use_namespaces:   cfg.use_namespaces,
   481→            read_only_rom:    cfg.read_only_rom,
   482→        };
   483→        if let Err(e) = mount_mgr::setup_mounts(&mount_cfg) {
   484→            error!("[KR64][child] mount setup failed: {}", e);
   485→            unsafe { libc::_exit(1) };
   486→        }
   487→
   488→        // Install seccomp filter on the guest process. This MUST be
   489→        // done after mount setup (otherwise the mount syscalls would
   490→        // be trapped and we couldn't set up the rootfs).
   491→        if cfg.install_seccomp {
   492→            info!("[KR64][child] installing seccomp filter");
   493→            if let Err(e) = seccomp::install() {
   494→                error!("[KR64][child] seccomp install failed: {}", e);
   495→                // Continue anyway — the guest will see the host's
   496→                // unfiltered syscalls, which is insecure but lets
   497→                // debugging proceed.
   498→            }
   499→        }
   500→
   501→        // Exec the guest init.
   502→        info!("[KR64][child] exec'ing guest init at {}", cfg.init_path);
   503→        let init_cstr = CString::new(cfg.init_path.as_str()).unwrap();
   504→        let argv0 = CString::new(cfg.init_path.as_str()).unwrap();
   505→        let argv: [*const libc::c_char; 2] = [argv0.as_ptr(), std::ptr::null()];
   506→        let envp: [*const libc::c_char; 1] = [std::ptr::null()];
   507→
   508→        // Set a few environment variables the guest init may look for.
   509→        // (execve replaces envp entirely, so we have to construct it.)
   510→        // For the MVP we just pass an empty envp — the guest's init
   511→        // builds its own environment from /proc/cmdline + properties.
   512→        let r = unsafe { libc::execve(init_cstr.as_ptr(), argv.as_ptr(), envp.as_ptr()) };
   513→        let e = std::io::Error::last_os_error();
   514→        error!("[KR64][child] execve({}) failed (r={}): {}", cfg.init_path, r, e);
   515→        unsafe { libc::_exit(127) };
   516→    }
   517→
   518→    // ----- PARENT (the daemon) -----
   519→    info!("[KR64][parent] guest pid = {}", pid);
   520→
   521→    // Spawn one accept thread per device socket. For the MVP each
   522→    // thread just accepts connections and immediately closes them
   523→    // (echoing a single byte so the guest sees SOME response). The
   524→    // production version will dispatch to per-device handlers:
   525→    //   qemu_pipe → openglrenderer::pipe
   526→    //   touch     → input::touch_server
   527→    //   key       → input::key_server
   528→    //   event     → TwoyiSocketServer (event IPC)
   529→    //   gb/gb2    → openglrenderer::gralloc
   530→    spawn_accept_thread(device_set.qemu_pipe, "qemu_pipe");
   531→    spawn_accept_thread(device_set.touch,     "touch");
   532→    spawn_accept_thread(device_set.key,       "key");
   533→    spawn_accept_thread(device_set.event,     "event");
   534→    spawn_accept_thread(device_set.gb.gb,     "gb");
   535→    spawn_accept_thread(device_set.gb.gb2,    "gb2");
   536→
   537→    // ---------------------------------------------------------------
   538→    // Step 6: wait for the guest to exit.
   539→    // ---------------------------------------------------------------
   540→    let mut status: libc::c_int = 0;
   541→    loop {
   542→        let waited = unsafe { libc::waitpid(pid, &mut status, 0) };
   543→        if waited == -1 {
   544→            let e = std::io::Error::last_os_error();
   545→            if e.raw_os_error() == Some(libc::EINTR) {
   546→                continue;
   547→            }
   548→            error!("[KR64][parent] waitpid failed: {}", e);
   549→            return 1;
   550→        }
   551→        break;
   552→    }
   553→
   554→    if libc::WIFEXITED(status) {
   555→        let code = libc::WEXITSTATUS(status);
   556→        info!("[KR64][parent] guest exited with status {}", code);
   557→        return code;
   558→    }
   559→    if libc::WIFSIGNALED(status) {
   560→        let sig = libc::WTERMSIG(status);
   561→        warning!("[KR64][parent] guest killed by signal {}", sig);
   562→        return 128 + sig;
   563→    }
   564→
   565→    warning!("[KR64][parent] guest waitpid returned unexpected status {}", status);
   566→    1
   567→}
   568→
   569→/// Spawn a thread that accepts connections on the given device socket
   570→/// and immediately closes them (MVP placeholder).
   571→///
   572→/// The thread takes ownership of the underlying `UnixListener` (so
   573→/// the parent `DeviceSocket` is consumed). For the production version
   574→/// this will be replaced with per-device handler dispatch.
   575→fn spawn_accept_thread(mut dev: devices::DeviceSocket, name: &'static str) {
   576→    let listener = match dev.take_listener() {
   577→        Some(l) => l,
   578→        None => {
   579→            warning!("[KR64] cannot spawn accept thread for {}: listener already taken", name);
   580→            return;
   581→        }
   582→    };
   583→    // Make the listening socket non-blocking so the thread can poll
   584→    // for shutdown signals (in the MVP we just loop forever).
   585→    let fd = listener.as_raw_fd();
   586→    let _ = unsafe { libc::fcntl(fd, libc::F_SETFL, libc::O_NONBLOCK) };
   587→
   588→    std::thread::Builder::new()
   589→        .name(format!("kr64-accept-{}", name))
   590→        .spawn(move || {
   591→            info!("[KR64][{}] accept thread started (fd={})", name, fd);
   592→            loop {
   593→                match listener.accept() {
   594→                    Ok((mut stream, _addr)) => {
   595→                        info!("[KR64][{}] client connected", name);
   596→                        // Echo a single byte so the guest sees SOME response.
   597→                        // (Many of the device protocols expect a handshake
   598→                        // byte — e.g. the touch device sends a device_info
   599→                        // struct on connect, which the guest reads before
   600→                        // sending anything. The production version will
   601→                        // dispatch to the right handler.)
   602→                        use std::io::Write;
   603→                        let _ = stream.write_all(&[0u8]);
   604→                    }
   605→                    Err(e) if e.kind() == std::io::ErrorKind::WouldBlock => {
   606→                        // No pending connection — sleep briefly to avoid
   607→                        // spinning. (Real implementation would use epoll.)
   608→                        std::thread::sleep(std::time::Duration::from_millis(50));
   609→                    }
   610→                    Err(e) => {
   611→                        warning!("[KR64][{}] accept error: {}", name, e);
   612→                        std::thread::sleep(std::time::Duration::from_millis(100));
   613→                    }
   614→                }
   615→            }
   616→        })
   617→        .expect("spawn kr64 accept thread");
   618→}

   620→// ============================================================================
   621→// cdylib entry point — used when `libkr64.so` is exec'd directly via
   622→// the PIE hack. The ELF entry point is set to `kr64_main` via the
   623→// `-Wl,-e,kr64_main` link flag in build.rs / .cargo/config.toml.
   624→// ============================================================================
   625→
   626→/// Entry point for the cdylib (libkr64.so). Mirrors the C `main`
   627→/// signature `(argc, argv) → int` so the standard C runtime can call
   628→/// it.
   629→///
   630→/// When the kernel exec's `libkr64.so`, the dynamic linker loads it,
   631→/// runs `.init_array`, and jumps to `kr64_main` (because we set
   632→/// `-Wl,-e,kr64_main`). The args come from the kernel's stack as
   633→/// `argc` + `argv[]`, just like a regular C `main`.
   634→#[no_mangle]
   635→pub extern "C" fn kr64_main(argc: libc::c_int, argv: *const *const libc::c_char) -> libc::c_int {
   636→    // Convert C argv → Rust Vec<String>.
   637→    let args: Vec<String> = if argc <= 0 || argv.is_null() {
   638→        Vec::new()
   639→    } else {
   640→        (0..argc as isize)
   641→            .map(|i| unsafe {
   642→                let p = *argv.offset(i);
   643→                if p.is_null() { String::new() }
   644→                else { std::ffi::CStr::from_ptr(p).to_string_lossy().to_string() }
   645→            })
   646→            .collect()
   647→    };
   648→
   649→    run(args)
   650→}

   652→// Reference the interp symbol from interp.c (built by build.rs) so the
   653→// linker keeps the `.interp` section in the final libkr64.so. Without
   654→// this reference, the linker may GC the `.interp` section as unused,
   655→// and the resulting .so won't be directly executable.
   656→extern "C" {
   657→    #[link_name = "interp"]
   658→    static INTERP: [u8; 0];
   659→}
   660→
   661→#[used]
   662→static INTERP_REF: &'static [u8; 0] = unsafe { &INTERP };
